You’re pointing to **/src/main.tsx** in `index.html`, but that file doesn’t exist. Create it (and a tiny shim for Auth if your context still lives under `client/`). Do exactly this:

# 1) Confirm entry

In **index.html**:

```html
<script type="module" src="/src/main.tsx"></script>
```

# 2) Create **/src/main.tsx**

```tsx
// /src/main.tsx
import { StrictMode, Suspense } from "react";
import { createRoot } from "react-dom/client";
import { createBrowserRouter, RouterProvider } from "react-router-dom";
import "./index.css";

import App from "./routes/App";
import Home from "./pages/home";
import Solutions from "./pages/services";
import Platform from "./pages/Platform";
import Industries from "./pages/Industries";
import Pricing from "./pages/pricing";
import RoiCenter from "./pages/RoiCenter";
import Resources from "./pages/Resources";
import About from "./pages/about";
import Contact from "./pages/contact";
import Login from "./pages/login";
import CustomerDashboard from "./pages/customer-dashboard";
// If you added this file, keep it; otherwise remove this import & the route below.
import ResourceDetail from "./pages/ResourceDetail";

import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { CartProvider } from "./contexts/CartContext";
import { AuthProvider } from "./contexts/AuthContext"; // shim below if not present
import PrivateRoute from "./ui/PrivateRoute";

const router = createBrowserRouter([
  {
    path: "/",
    element: <App />,
    children: [
      { index: true, element: <Home /> },
      { path: "solutions", element: <Solutions /> },
      { path: "platform", element: <Platform /> },
      { path: "industries", element: <Industries /> },
      { path: "pricing", element: <Pricing /> },
      { path: "roi", element: <RoiCenter /> },
      { path: "resources", element: <Resources /> },
      { path: "resources/:slug", element: <ResourceDetail /> }, // remove if file not present
      { path: "about", element: <About /> },
      { path: "contact", element: <Contact /> },
      { path: "login", element: <Login /> },
      {
        path: "customer-dashboard",
        element: (
          <PrivateRoute>
            <CustomerDashboard />
          </PrivateRoute>
        ),
      },
      { path: "*", element: <div className="p-8 text-center">404 — Not found</div> },
    ],
  },
]);

const qc = new QueryClient();

createRoot(document.getElementById("root")!).render(
  <StrictMode>
    <QueryClientProvider client={qc}>
      <AuthProvider>
        <CartProvider>
          <Suspense fallback={<div className="p-6">Loading…</div>}>
            <RouterProvider router={router} />
          </Suspense>
        </CartProvider>
      </AuthProvider>
    </QueryClientProvider>
  </StrictMode>
);
```

# 3) If **/src/contexts/AuthContext.tsx** doesn’t exist, add this shim

```ts
// /src/contexts/AuthContext.tsx
export * from "../../client/src/contexts/AuthContext";
```

# 4) If **/src/ui/PrivateRoute.tsx** is missing, add:

```tsx
// /src/ui/PrivateRoute.tsx
import { Navigate, useLocation } from "react-router-dom";
import { useAuth } from "../contexts/AuthContext";
export default function PrivateRoute({ children }: { children: React.ReactNode }) {
  const { isAuthenticated } = useAuth();
  const loc = useLocation();
  if (!isAuthenticated) return <Navigate to="/login" state={{ from: loc.pathname }} replace />;
  return <>{children}</>;
}
```

# 5) Restart

In the shell:

```bash
npm i @tanstack/react-query react-router-dom
npm run dev
```

This removes the “Failed to load url /src/main.tsx” error and keeps a single provider chain so `useAuth`, React Query, and your routes work without crashing.
