```css
  }

  .btn-vs:active { background: var(--vs-purple-700); }
  .btn-vs:focus { @apply ring-[var(--vs-purple)] ring-offset-white; }

  /* Outline variant (for secondary) */
  .btn-vs-outline {
    @apply inline-flex items-center justify-center rounded-md px-4 py-2 text-sm font-semibold transition-colors border;
    color: var(--vs-purple);
    border-color: var(--vs-purple);
  }
  .btn-vs-outline:hover {
    color: #fff;
    background: var(--vs-purple);
    border-color: var(--vs-purple);
  }
}
```
