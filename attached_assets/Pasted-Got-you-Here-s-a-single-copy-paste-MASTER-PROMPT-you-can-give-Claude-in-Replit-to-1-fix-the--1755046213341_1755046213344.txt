6. Auth middleware — **reads `Authorization: Bearer` header OR cookie** so it works in the Replit iframe where third-party cookies are blocked:

**server/middleware/auth.ts**

```ts
import type { Request, Response, NextFunction } from 'express';
import { verify } from '../lib/jwt';

export function extractToken(req: Request){
  const h = req.headers['authorization'];
  if (h && typeof h === 'string' && h.startsWith('Bearer ')) return h.slice(7);
  // fallback cookie:
  // @ts-ignore
  const c = req.cookies?.vs_token;
  return c;
}

export function requireAuth(req: Request, res: Response, next: NextFunction){
  const t = extractToken(req); if (!t) return res.status(401).json({message:'Unauthenticated'});
  try { (req as any).user = verify(t); next(); } catch { return res.status(401).json({message:'Invalid session'}); }
}

export function requireRole(...roles: string[]){
  return (req:any,res:Response,next:NextFunction)=> roles.includes(req.user?.role) ? next() : res.status(403).json({message:'Forbidden'});
}
```

export async function api(path:string, init?:RequestInit){

import React, { useState } from 'react';
import { api } from '@/api/client';
import { useNavigate } from 'react-router-dom';

export default function Register(){
  const nav = useNavigate();
  const [stage,setStage]=useState<'start'|'verify'>('start');
  const [p,setP]=useState({ code:'', email:'', phone:'', full_name:'', org_name:'', password:'' });
  const [token,setToken]=useState(''); const [emailCode,setEmailCode]=useState(''); const [smsCode,setSmsCode]=useState(''); const [err,setErr]=useState('');

  async function start(e:React.FormEvent){ e.preventDefault(); setErr(''); const r = await api('/auth/register/start',{ method:'POST', body:JSON.stringify(p) }); setToken(r.token); setStage('verify'); }
  async function verify(e:React.FormEvent){ e.preventDefault(); setErr('');
    const r = await api('/auth/register/verify',{ method:'POST', body:JSON.stringify({ token, code:p.code, email:p.email, phone:p.phone, full_name:p.full_name, org_name:p.org_name, password:p.password, email_code:emailCode, sms_code:smsCode }) });
    localStorage.setItem('vs_token', r.token);
    nav('/admin');
  }

  return (
    <main className="max-w-xl mx-auto p-6">
      <h1 className="text-2xl font-semibold mb-4">Create your VictorySync account</h1>
      {stage==='start' ? (
        <form onSubmit={start} className="space-y-3">
          <input className="input" placeholder="Access code" value={p.code} onChange={e=>setP({...p,code:e.target.value})} required/>
          <input className="input" placeholder="Full name" value={p.full_name} onChange={e=>setP({...p,full_name:e.target.value})} required/>
          <input className="input" placeholder="Organization" value={p.org_name} onChange={e=>setP({...p,org_name:e.target.value})} required/>

        </form>
      ) : (
        <form onSubmit={verify} className="space-y-3">
          <input className="input" placeholder="Email code" value={emailCode} onChange={e=>setEmailCode(e.target.value)} required/>

import React, { useEffect, useState } from 'react';
import { api } from '@/api/client';

type UserRow = {
  id:string; email:string; phone:string; full_name:string; role:string; is_active:boolean;
  email_verified:boolean; phone_verified:boolean; org_name:string|null;
  can_use_api:boolean; can_manage_org:boolean; can_view_billing:boolean; integrations:Record<string,boolean>;
  allow_card:boolean; allow_paypal:boolean; allow_wire:boolean;
};

export default function AdminPortal(){
  const [users,setUsers]=useState<UserRow[]>([]); const [loading,setLoading]=useState(true); const [error,setError]=useState('');

  async function load(){
    try{ setLoading(true); setUsers(await api('/admin/users')); }
    catch(e:any){ setError(e.message); }
    finally{ setLoading(false); }
  }
  useEffect(()=>{ load(); },[]);

  async function save(u:UserRow){
    await api(`/admin/users/${u.id}/policy`,{ method:'POST', body:JSON.stringify({
      can_use_api:u.can_use_api, can_manage_org:u.can_manage_org, can_view_billing:u.can_view_billing,
      integrations:u.integrations, allow_card:u.allow_card, allow_paypal:u.allow_paypal, allow_wire:u.allow_wire, is_active:u.is_active
    })});
  }

  if (error && /Unauthenticated|Invalid session|Forbidden/i.test(error)) {
    return <main className="max-w-4xl mx-auto p-6"><h1 className="text-2xl font-semibold">Admin Portal</h1><p className="text-red-600 mt-3">Unauthenticated</p><a className="btn btn-primary mt-3 inline-block" href="/login">Sign in</a></main>;
  }

  return (
    <main className="max-w-6xl mx-auto p-6">
      <h1 className="text-2xl font-semibold mb-4">Admin Portal</h1>
      {loading && <p>Loading…</p>}
      {!loading && users.length===0 && <p>No users found.</p>}
      {!loading && users.map(u=>(
        <div key={u.id} className="rounded border p-4 mb-3 bg-white">
          <div className="font-medium">{u.full_name} <span className="text-gray-500">({u.email})</span> — {u.org_name||'No org'}</div>
          <div className="grid grid-cols-2 md:grid-cols-4 gap-2 my-3">
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.can_use_api} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,can_use_api:e.target.checked}:x))}/> API</label>
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.can_manage_org} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,can_manage_org:e.target.checked}:x))}/> Manage Org</label>
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.can_view_billing} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,can_view_billing:e.target.checked}:x))}/> View Billing</label>
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.is_active} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,is_active:e.target.checked}:x))}/> Active</label>
          </div>
          <div className="grid grid-cols-2 md:grid-cols-3 gap-2 my-3">
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.allow_card} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,allow_card:e.target.checked}:x))}/> Card</label>
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.allow_paypal} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,allow_paypal:e.target.checked}:x))}/> PayPal</label>
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.allow_wire} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,allow_wire:e.target.checked}:x))}/> Wire</label>
          </div>
          <div className="grid grid-cols-2 md:grid-cols-4 gap-2 my-3">
            {['hubspot','salesforce','zohocrm','five9','talkdesk','gohighlevel'].map(k=>(
              <label key={k} className="flex items-center gap-2">
                <input type="checkbox" checked={!!u.integrations?.[k]} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,integrations:{...x.integrations,[k]:e.target.checked}}:x))}/>{k}
              </label>
            ))}
          </div>
          <button className="btn btn-primary" onClick={()=>save(u)}>Save</button>
        </div>
      ))}
    </main>
  );
}
