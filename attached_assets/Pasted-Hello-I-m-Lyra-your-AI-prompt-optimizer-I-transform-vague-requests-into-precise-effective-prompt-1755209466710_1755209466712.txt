## 3) Fix `/pricing` crash (Cart provider)

### Edit: `client/src/context/cart.tsx`

```tsx
import React, { createContext, useContext, useState } from "react";
type Item = { id: string; qty: number };
type Ctx = { items: Item[]; addItem: (id: string, qty?: number) => void; clear: () => void; };
const C = createContext<Ctx | undefined>(undefined);

export const CartProvider: React.FC<React.PropsWithChildren> = ({ children }) => {
  const [items, set] = useState<Item[]>([]);
  const addItem = (id: string, qty = 1) =>
    set(prev => {
      const i = prev.findIndex(x => x.id === id);
      if (i > -1) { const copy = [...prev]; copy[i] = { ...copy[i], qty: copy[i].qty + qty }; return copy; }
      return [...prev, { id, qty }];
    });
  return <C.Provider value={{ items, addItem, clear: () => set([]) }}>{children}</C.Provider>;
};

export const useCartContext = () => {
  const v = useContext(C);
  if (!v) throw new Error("useCartContext must be used within a CartProvider");
  return v;
};
```
