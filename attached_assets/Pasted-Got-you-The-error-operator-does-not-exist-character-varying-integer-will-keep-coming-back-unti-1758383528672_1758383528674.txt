Got you. The error `operator does not exist: character varying = integer` will keep coming back until we **normalize your schema**: right now `organizations.id` is a **text/varchar**, while the rest of the app (joins, `users.org_id`, `entitlements.org_id`) expects an **integer**. That mismatch is what breaks log-in and entitlement lookups.

Here’s a permanent, production-safe fix:

---

# Explanation

* Convert `organizations.id` to a **real integer primary key**.
* Keep your legacy row (`id='demo-scale'`) but move it to a `_legacy` column so it can’t break joins.
* Make sure `entitlements.value` exists (your code reads it).
* Upsert a **numeric** “Scale” org and a **super-admin** demo user, then grant **all** features.

This removes all varchar↔integer comparisons for good.

# Code

### 1) One-time DB migration (run in Neon SQL console)

```sql
BEGIN;

-- 0) Make sure 'value' exists on entitlements (your code reads it)
CREATE TABLE IF NOT EXISTS entitlements (
  id         SERIAL PRIMARY KEY,
  org_id     INTEGER NOT NULL,
  feature_id INTEGER NOT NULL,
  enabled    BOOLEAN NOT NULL DEFAULT TRUE
);
ALTER TABLE entitlements
  ADD COLUMN IF NOT EXISTS value TEXT;
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conname='entitlements_org_feature_key'
  ) THEN
    ALTER TABLE entitlements ADD CONSTRAINT entitlements_org_feature_key UNIQUE (org_id, feature_id);
  END IF;
END $$;

-- 1) Ensure features table exists (used for Scale entitlements)
CREATE TABLE IF NOT EXISTS features (
  id   SERIAL PRIMARY KEY,
  code VARCHAR(64) UNIQUE NOT NULL,
  name VARCHAR(128)        NOT NULL
);
INSERT INTO features (code, name) VALUES
  ('blended_team',        '24/7 Blended Team'),
  ('ai_suite',            'AI Automation Suite'),
  ('success_manager',     'Dedicated Success Manager'),
  ('custom_integrations', 'Custom Integrations'),
  ('priority_support',    'Priority Support'),
  ('realtime_analytics',  'Real-time Analytics')
ON CONFLICT (code) DO NOTHING;

-- 2) Organizations: add columns your app uses
ALTER TABLE organizations
  ADD COLUMN IF NOT EXISTS name       VARCHAR(255),
  ADD COLUMN IF NOT EXISTS plan_id    VARCHAR(32),
  ADD COLUMN IF NOT EXISTS timezone   VARCHAR(64) DEFAULT 'America/New_York',
  ADD COLUMN IF NOT EXISTS created_at TIMESTAMP    DEFAULT NOW();

-- 3) Create an INTEGER id and move PK to it (keep the old text id as 'id_legacy')
--    Works whether current id is text or int.
ALTER TABLE organizations
  ADD COLUMN IF NOT EXISTS id_int BIGINT GENERATED BY DEFAULT AS IDENTITY;

-- For numeric-looking legacy ids, preserve value; otherwise keep generated
UPDATE organizations
SET id_int = CASE
  WHEN id::text ~ '^[0-9]+$' THEN (id::text)::bigint
  ELSE id_int
END;

-- Not null & PK swap
ALTER TABLE organizations ALTER COLUMN id_int SET NOT NULL;
DO $$
BEGIN
  IF EXISTS (
    SELECT 1 FROM information_schema.table_constraints
    WHERE table_name='organizations' AND constraint_type='PRIMARY KEY'
  ) THEN
    ALTER TABLE organizations DROP CONSTRAINT organizations_pkey;
  END IF;
END $$;
ALTER TABLE organizations ADD CONSTRAINT organizations_pkey PRIMARY KEY (id_int);

-- Keep old id for audit, switch app to integer id
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM information_schema.columns
    WHERE table_name='organizations' AND column_name='id_legacy'
  ) THEN
    ALTER TABLE organizations RENAME COLUMN id TO id_legacy;
  END IF;
END $$;
ALTER TABLE organizations RENAME COLUMN id_int TO id;

-- Avoid picking legacy string-id rows by name in the future
UPDATE organizations
SET name = name || ' [legacy string-id — ignore]'
WHERE id_legacy IS NOT NULL AND id_legacy::text !~ '^[0-9]+$'
  AND name NOT LIKE '%[legacy string-id — ignore]%';

-- 4) Make org names unique (deterministic upserts)
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_indexes
    WHERE schemaname='public' AND indexname='organizations_name_idx'
  ) THEN
    CREATE UNIQUE INDEX organizations_name_idx ON organizations (name);
  END IF;
END $$;

-- 5) Ensure public_users table (used by login)
CREATE TABLE IF NOT EXISTS public_users (
  id             SERIAL PRIMARY KEY,
  name           VARCHAR(120) NOT NULL,
  email          VARCHAR(255) NOT NULL UNIQUE,
  password_hash  VARCHAR(255) NOT NULL,
  provider       VARCHAR(32)  NOT NULL DEFAULT 'local',
  provider_id    VARCHAR(128),
  social_accounts JSONB       NOT NULL DEFAULT '[]'::jsonb,
  privileges     JSONB        NOT NULL DEFAULT '{}'::jsonb,
  created_at     TIMESTAMP    NOT NULL DEFAULT NOW()
);
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_indexes
    WHERE schemaname='public' AND indexname='public_users_email_idx'
  ) THEN
    CREATE UNIQUE INDEX public_users_email_idx ON public_users (email);
  END IF;
END $$;

-- 6) Optional users table (if your code uses it)
CREATE TABLE IF NOT EXISTS users (
  id                SERIAL PRIMARY KEY,
  org_id            INTEGER,
  name              VARCHAR(120) NOT NULL,
  email             VARCHAR(255) NOT NULL UNIQUE,
  password_hash     VARCHAR(255) NOT NULL,
  two_factor_enabled BOOLEAN     NOT NULL DEFAULT FALSE,
  two_factor_secret  TEXT,
  is_super_admin     BOOLEAN     NOT NULL DEFAULT FALSE,
  is_active          BOOLEAN     NOT NULL DEFAULT TRUE,
  created_at         TIMESTAMP   NOT NULL DEFAULT NOW()
);

COMMIT;
```

### 2) Final seeder (works with the normalized schema)

Create `scripts/seedDemoScale.mjs` (replace if you already have one):

```js
import { neon, neonConfig } from "@neondatabase/serverless";
import WebSocket from "ws";
import bcrypt from "bcryptjs";

neonConfig.webSocketConstructor = WebSocket;
const sql = neon(process.env.DATABASE_URL);

const EMAIL  = "demo@victorysync.com";
const PASS   = "VictoryScale123!";
const NAME   = "Demo Admin";

const ORG    = "VictorySync Demo (Scale)";
const ORG_TZ = "America/New_York";
const PLAN   = "scale";

async function upsertOrgReturnId() {
  // Use unique org name; id is now INTEGER PK
  const r = await sql/*sql*/`
    INSERT INTO organizations (name, timezone, plan_id)
    VALUES (${ORG}, ${ORG_TZ}, ${PLAN})
    ON CONFLICT (name) DO UPDATE
      SET plan_id = EXCLUDED.plan_id, timezone = EXCLUDED.timezone
    RETURNING id
  `;
  return Number(r[0].id);
}

async function main() {
  if (!process.env.DATABASE_URL) throw new Error("DATABASE_URL not set");
  if (!process.env.JWT_SECRET)   throw new Error("JWT_SECRET not set");

  console.log("STEP 1: ensure Scale org (integer id) …");
  const orgId = await upsertOrgReturnId();
  console.log("  orgId:", orgId);

  console.log("STEP 2: upsert demo admin in public_users …");
  const hash = await bcrypt.hash(PASS, 10);
  await sql/*sql*/`
    INSERT INTO public_users (name, email, password_hash, provider, provider_id, social_accounts, privileges)
    VALUES (${NAME}, lower(${EMAIL}), ${hash}, 'local', NULL, '[]'::jsonb, '{"role":"super_admin"}'::jsonb)
    ON CONFLICT (email) DO UPDATE
      SET name=EXCLUDED.name,
          password_hash=EXCLUDED.password_hash,
          provider='local',
          provider_id=NULL,
          social_accounts='[]'::jsonb,
          privileges='{"role":"super_admin"}'::jsonb,
          email=lower(${EMAIL})
  `;

  console.log("STEP 3: upsert into users (if table exists) …");
  try {
    await sql/*sql*/`
      INSERT INTO users (org_id, name, email, password_hash, two_factor_enabled, is_super_admin, is_active)
      VALUES (${orgId}, ${NAME}, lower(${EMAIL}), ${hash}, false, true, true)
      ON CONFLICT (email) DO UPDATE
        SET org_id=${orgId},
            name=EXCLUDED.name,
            password_hash=EXCLUDED.password_hash,
            two_factor_enabled=false,
            is_super_admin=true,
            is_active=true,
            email=lower(${EMAIL})
    `;
  } catch (e) {
    if (!String(e?.message||e).includes('relation "users" does not exist')) throw e;
    console.log("  users table not found; skipping.");
  }

  console.log("STEP 4: grant ALL features to org (covers Scale) …");
  await sql/*sql*/`
    INSERT INTO entitlements (org_id, feature_id, enabled, value)
    SELECT ${orgId}, f.id, TRUE, NULL::text
    FROM features f
    WHERE NOT EXISTS (
      SELECT 1 FROM entitlements e
      WHERE e.org_id=${orgId} AND e.feature_id=f.id
    )
  `;

  console.log("✅ Demo ready. Login:", EMAIL, PASS, "Org:", ORG, "OrgId:", orgId);
}

main().catch(e => { console.error("❌ seedDemoScale failed:", e?.message || e); process.exit(1); });
```
