**Create/replace:** `client/src/pages/SignUp.tsx`  (access-code → verify → OTP)

```tsx
import { useState } from "react";

type VerifyPayload = {
  accessCode: string;
  name: string;
  email: string;
  phone: string;
  orgNumber: string;
};

export default function SignUp() {
  const [step, setStep] = useState<"verify"|"otp"|"done">("verify");
  const [form, setForm] = useState<VerifyPayload>({ accessCode:"", name:"", email:"", phone:"", orgNumber:"" });
  const [otp, setOtp] = useState("");
  const [error, setError] = useState<string | null>(null);
  const [debugOtp, setDebugOtp] = useState<string | null>(null); // shows in dev

  const onChange = (k: keyof VerifyPayload) => (e: any) => setForm({ ...form, [k]: e.target.value });

  async function verify() {
    setError(null);
    const res = await fetch("/api/access/verify", {
      method:"POST", headers:{ "Content-Type":"application/json" },
      body: JSON.stringify(form)
    });
    const data = await res.json();
    if (!res.ok || !data.ok) { setError(data.error || "Verification failed"); return; }

    const res2 = await fetch("/api/access/request-otp", {
      method:"POST", headers:{ "Content-Type":"application/json" },
      body: JSON.stringify({ accessCode: form.accessCode, email: form.email, phone: form.phone })
    });
    const data2 = await res2.json();
    if (!res2.ok || !data2.ok) { setError(data2.error || "Could not send OTP"); return; }

    if (data2.devOtp) setDebugOtp(data2.devOtp); // shown only in dev
    setStep("otp");
  }

  async function confirmOtp(e: any) {
    e.preventDefault();
    setError(null);
    const res = await fetch("/api/access/confirm-otp", {
      method:"POST", headers:{ "Content-Type":"application/json" },
      body: JSON.stringify({ accessCode: form.accessCode, email: form.email, otp })
    });
    const data = await res.json();
    if (!res.ok || !data.ok) { setError(data.error || "Invalid OTP"); return; }
    setStep("done");
  }

  if (step === "done") {
    return (
      <section className="max-w-md mx-auto p-6">
        <h1 className="text-2xl font-bold mb-2">Account Verified</h1>
        <p className="mb-4">Your access code was verified and OTP confirmed.</p>
        <a className="underline" href="/signin">Continue to Sign In</a>
      </section>
    );
  }

  return (
    <section className="max-w-md mx-auto p-6">
      <h1 className="text-2xl font-bold mb-4">Create Your Account</h1>

      {step === "verify" && (
        <form className="space-y-3" onSubmit={(e)=>{e.preventDefault(); verify();}}>
          <input className="w-full border p-2 rounded" placeholder="Access Code" value={form.accessCode} onChange={onChange("accessCode")} required />
          <input className="w-full border p-2 rounded" placeholder="Full Name" value={form.name} onChange={onChange("name")} required />
          <input className="w-full border p-2 rounded" placeholder="Work Email" value={form.email} onChange={onChange("email")} required />
          <input className="w-full border p-2 rounded" placeholder="Phone Number" value={form.phone} onChange={onChange("phone")} required />
          <input className="w-full border p-2 rounded" placeholder="Organization Number" value={form.orgNumber} onChange={onChange("orgNumber")} required />
          {error && <p className="text-red-600 text-sm">{error}</p>}
          <button className="w-full py-2 rounded bg-black text-white">Verify & Send OTP</button>
        </form>
      )}

      {step === "otp" && (
        <form className="space-y-3" onSubmit={confirmOtp}>
          <input className="w-full border p-2 rounded" placeholder="Enter OTP" value={otp} onChange={(e)=>setOtp(e.target.value)} required />
          {debugOtp && <p className="text-xs text-gray-500">Dev OTP: {debugOtp}</p>}
          {error && <p className="text-red-600 text-sm">{error}</p>}
          <button className="w-full py-2 rounded bg-black text-white">Confirm OTP</button>
        </form>
      )}
    </section>
  );
}
```

```ts
// ===== ACCESS-CODE SIGNUP API =====
import fs from "fs";
type AccessCode = { code: string; orgNumber: string; orgName?: string; emailDomain?: string };
type OtpEntry = { key: string; otp: string; expiresAt: number }; // key = `${code}|${email}`

const CODES_PATH = path.join(__dirname, "data", "access-codes.json");
const OTPS_PATH  = path.join(__dirname, "data", "otps.json");

function readJson<T>(p: string, fallback: T): T {
  try { return JSON.parse(fs.readFileSync(p, "utf8")) as T; }
  catch { return fallback; }
}
function writeJson<T>(p: string, data: T) {
  fs.mkdirSync(path.dirname(p), { recursive: true });
  fs.writeFileSync(p, JSON.stringify(data, null, 2));
}
function genOtp(len = 6) {
  return Array.from({ length: len }, () => Math.floor(Math.random()*10)).join("");
}

// Verify access code + user/org details
api.post("/access/verify", (req, res) => {
  const { accessCode, name, email, phone, orgNumber } = req.body ?? {};
  if (!accessCode || !name || !email || !phone || !orgNumber)
    return res.status(400).json({ ok:false, error:"Missing required fields" });

  const codes = readJson<AccessCode[]>(CODES_PATH, []);
  const found = codes.find(c => c.code === accessCode);
  if (!found) return res.status(404).json({ ok:false, error:"Invalid access code" });

  if (found.orgNumber && String(found.orgNumber) !== String(orgNumber))
    return res.status(400).json({ ok:false, error:"Organization number mismatch" });

  if (found.emailDomain && !String(email).toLowerCase().endsWith(`@${found.emailDomain.toLowerCase()}`))
    return res.status(400).json({ ok:false, error:`Email must end with @${found.emailDomain}` });

  return res.json({ ok:true, orgName: found.orgName ?? null });
});

// Send/request OTP
api.post("/access/request-otp", (req, res) => {
  const { accessCode, email, phone } = req.body ?? {};
  if (!accessCode || !email) return res.status(400).json({ ok:false, error:"Missing accessCode or email" });

  const otps = readJson<OtpEntry[]>(OTPS_PATH, []);
  const key = `${accessCode}|${email.toLowerCase()}`;
  const otp = genOtp(6);
  const expiresAt = Date.now() + 5 * 60 * 1000; // 5 minutes

  const idx = otps.findIndex(o => o.key === key);
  if (idx > -1) otps[idx] = { key, otp, expiresAt }; else otps.push({ key, otp, expiresAt });
  writeJson(OTPS_PATH, otps);

  // "Send" OTP — integrate real SMS/Email later. For now, expose in dev only.
  const devOtp = process.env.NODE_ENV !== "production" ? otp : undefined;
  console.log(`[OTP] Sent OTP ${otp} to ${email}${phone ? " / " + phone : ""} for ${accessCode}`);

  return res.json({ ok:true, devOtp });
});

// Confirm OTP
api.post("/access/confirm-otp", (req, res) => {
  const { accessCode, email, otp } = req.body ?? {};
  if (!accessCode || !email || !otp) return res.status(400).json({ ok:false, error:"Missing fields" });

  const otps = readJson<OtpEntry[]>(OTPS_PATH, []);
  const key = `${accessCode}|${email.toLowerCase()}`;
  const entry = otps.find(o => o.key === key);
  if (!entry) return res.status(400).json({ ok:false, error:"No OTP pending" });
  if (Date.now() > entry.expiresAt) return res.status(400).json({ ok:false, error:"OTP expired" });
  if (String(entry.otp) !== String(otp)) return res.status(400).json({ ok:false, error:"Invalid OTP" });

  // Remove OTP after success
  writeJson(OTPS_PATH, otps.filter(o => o.key !== key));

  // TODO: Create user in your DB here.
  return res.json({ ok:true });
});
```
