import 'dotenv/config'
import express from 'express'
import path from 'node:path'
import { fileURLToPath } from 'node:url'

const __filename = fileURLToPath(import.meta.url)
const __dirname = path.dirname(__filename)

const app = express()

// Basic hardening & json
app.disable('x-powered-by')
app.use(express.json({ limit: '1mb' }))
app.use(express.urlencoded({ extended: true }))

// Health check for Replit
app.get('/health', (_req, res) => {
  res.json({ ok: true, env: process.env.NODE_ENV || 'development' })
})

// Mount your existing API routes BEFORE SPA handler
// import { router as authRouter } from './routes/auth.js'
// app.use('/api/auth', authRouter)
// ... mount other routes
// (IMPORTANT: ensure these imports are valid paths; fix any broken imports)

const isProd = process.env.NODE_ENV === 'production'
const PORT = Number(process.env.PORT) || 3000
const HOST = '0.0.0.0'

async function start() {
  if (!isProd) {
    // DEV: Use Vite in middleware mode on SAME Express port
    const { createServer: createViteServer } = await import('vite')
    const vite = await createViteServer({
      root: path.resolve(__dirname, '../..', 'client'),
      server: {
        middlewareMode: true,
        hmr: { port: 24678 },
        watch: { usePolling: true, interval: 150 }
      },
      appType: 'custom'
    })

    app.use(vite.middlewares)

    // Let Vite handle index.html and route fallback
    app.use('*', async (req, res) => {
      try {
        const url = req.originalUrl
        const template = await vite.transformIndexHtml(url, '')
        res.status(200).set({ 'Content-Type': 'text/html' }).end(template)
      } catch (e: any) {
        vite.ssrFixStacktrace(e)
        console.error(e)
        res.status(500).end(e.message)
      }
    })
  } else {
    // PROD: serve built client
    const clientDist = path.resolve(__dirname, '../../client/dist')
    app.use(express.static(clientDist))
    app.get('*', (_req, res) => {
      res.sendFile(path.join(clientDist, 'index.html'))
    })
  }

  app.listen(PORT, HOST, () => {
    console.log(`✅ Server ready on http://${HOST}:${PORT} (NODE_ENV=${process.env.NODE_ENV})`)
  })
}

start().catch((err) => {
  console.error('Failed to start server:', err)
  process.exit(1)
})
