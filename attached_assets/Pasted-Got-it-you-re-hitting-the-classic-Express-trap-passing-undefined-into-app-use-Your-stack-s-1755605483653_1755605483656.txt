### ✅ Task 2 — Replace `server/index.ts` with a safe, production-ready bootstrap

**Why:** Your stack points to line 57 inside `server/index.ts`. That almost always means `app.use()` received `undefined` (e.g., `app.use(isProduction && morgan('tiny'))`, or an array like `app.use([cors(), helmet(), maybeUndefined])`). Express can’t accept `undefined` middleware.

**Fix:** Use explicit guards (no boolean short-circuit) and never pass arrays to `app.use()` that might contain `undefined`.

```ts
// server/index.ts
import path from 'node:path';
import express from 'express';
import cors from 'cors';
import helmet from 'helmet';
import compression from 'compression';
import cookieParser from 'cookie-parser';
import morgan from 'morgan';
import routes from './routes';

const app = express();

// ---------- ENV ----------
const NODE_ENV = process.env.NODE_ENV ?? 'development';
const isProd = NODE_ENV === 'production';

// ---------- CORE MIDDLEWARE (NO UNDEFINED PASSED) ----------
app.set('trust proxy', 1);
app.use(express.json({ limit: '1mb' }));
app.use(express.urlencoded({ extended: true }));
app.use(cookieParser());

// CORS (adjust origins as needed; leaving permissive for now)
app.use(
  cors({
    origin: true,
    credentials: true,
  })
);

// Helmet (safe defaults)
app.use(
  helmet({
    crossOriginResourcePolicy: false, // allows static assets if needed
  })
);

// Compression
app.use(compression());

// Logging — guard explicitly so we never pass undefined
if (isProd) {
  app.use(morgan('tiny'));
} else {
  app.use(morgan('dev'));
}

// ---------- API ROUTES ----------
app.use('/api', routes);

// ---------- STATIC + SPA FALLBACK ----------
const distDir = path.resolve(__dirname, '../client/dist');
app.use(express.static(distDir));

// Health checks
app.get('/healthz', (_req, res) => res.json({ ok: true, env: NODE_ENV }));

// Send index.html for any non-API route (SPA fallback)
app.get('*', (req, res, next) => {
  if (req.path.startsWith('/api')) return next();
  res.sendFile(path.join(distDir, 'index.html'));
});

export default app;
```

> Notes:
>
> * No `app.use(isProd && morgan('tiny'))` or `app.use([foo, maybeUndefined])`.
> * Everything is guarded with `if` to avoid passing `undefined`.
