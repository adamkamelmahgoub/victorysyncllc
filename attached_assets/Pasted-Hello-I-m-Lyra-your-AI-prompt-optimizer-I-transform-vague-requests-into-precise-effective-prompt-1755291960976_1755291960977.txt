## STEP 1 — Create the **only** auth context (safe hook)

### **REPLACE/CREATE** `client/src/context/auth.tsx`

```tsx
import { createContext, useContext, useEffect, useState, PropsWithChildren } from "react";

type User = { id?: string; email?: string; name?: string } | null;

type AuthCtx = {
  user: User;
  authenticated: boolean;
  loading: boolean;
  error?: string;
  refresh: () => void;
  login: () => Promise<void>;
  logout: () => Promise<void>;
};

const AuthContext = createContext<AuthCtx | undefined>(undefined);

export function AuthProvider({ children }: PropsWithChildren) {
  const [user, setUser] = useState<User>(null);
  const [authenticated, setAuthed] = useState(false);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | undefined>(undefined);

  const load = async () => {
    try {
      setLoading(true);
      const res = await fetch("/api/auth/check", { credentials: "include" });
      if (!res.ok) throw new Error(`${res.status} ${res.statusText}`);
      const data = await res.json().catch(() => ({ authenticated: false, user: null }));
      setUser(data?.user ?? null);
      setAuthed(Boolean(data?.authenticated));
      setError(undefined);
    } catch (e: any) {
      setUser(null);
      setAuthed(false);
      setError(String(e?.message || e));
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => { load(); }, []);

  return (
    <AuthContext.Provider value={{
      user, authenticated, loading, error,
      refresh: load,
      login: async () => {},
      logout: async () => {}
    }}>
      {children}
    </AuthContext.Provider>
  );
}

/** SAFE HOOK: never crash the app if provider is missing */
export function useAuth(): AuthCtx {
  const ctx = useContext(AuthContext);
  if (!ctx) {
    console.warn("useAuth called without AuthProvider — returning safe defaults.");
    return {
      user: null,
      authenticated: false,
      loading: false,
      error: "AuthProvider missing",
      refresh: () => {},
      login: async () => {},
      logout: async () => {},
    };
  }
  return ctx;
}
```

## STEP 5 — File-system router renders **all** pages

### **REPLACE** `client/src/router.tsx` (auto-register every page under `/pages`)

```tsx
import { lazy, Suspense } from "react";
import { Routes, Route, Navigate } from "react-router-dom";
import RootLayout from "@/layouts/RootLayout";
import NotFound from "@/pages/NotFound";

type Loader = () => Promise<{ default: React.ComponentType<any> }>;
const pageModules = import.meta.glob("./pages/**/*.tsx") as Record<string, Loader>;

function fileToPath(file: string): string {
  let p = file.replace("./pages", "").replace(/\.tsx$/, "");
  p = p.replace(/\/?Index$/i, "");
  if (/^\/Home$/i.test(p)) p = "";
  p = p.replace(/\[([^\]]+)\]/g, ":$1");
  p = (p ? p : "/").toLowerCase();
  return p;
}

const autoRoutes = Object.entries(pageModules)
  .map(([file, loader]) => ({ path: fileToPath(file), Component: lazy(loader) }))
  .filter(r => r.path !== "/404" && r.path !== "/notfound");

export default function AppRouter() {
  return (
    <Suspense fallback={null}>
      <Routes>
        <Route element={<RootLayout />}>
          {autoRoutes
            .filter(r => r.path === "/")
            .map(({ path, Component }) => <Route key={path} index element={<Component />} />)}
          {autoRoutes
            .filter(r => r.path !== "/")
            .map(({ path, Component }) => <Route key={path} path={path.slice(1)} element={<Component />} />)}
          <Route path="404" element={<NotFound />} />
          <Route path="*" element={<Navigate to="/404" replace />} />
        </Route>
      </Routes>
    </Suspense>
  );
}
```

*(Ensure each page under `client/src/pages/**` has a **default export** component.)*
