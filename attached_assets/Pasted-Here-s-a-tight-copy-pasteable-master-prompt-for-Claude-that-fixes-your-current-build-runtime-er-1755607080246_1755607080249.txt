router.post('/forgot-password', async (req, res) => {
  const parse = forgotSchema.safeParse(req.body)
  if (!parse.success) return res.status(400).json({ error: 'Invalid payload' })

const resetSchema = z.object({
  token: z.string().min(10),
  password: z.string().min(8)
})

  const parse = resetSchema.safeParse(req.body)
  if (!parse.success) return res.status(400).json({ error: 'Invalid payload' })

  const row = await db.passwordResets.findByToken(token)
  if (!row || row.expiresAt < new Date()) {

  await db.user.updatePassword(row.userId, password) // use argon2/bcrypt as in your auth

### Step 4 — **Forgot Password screens (Customer + Admin)**

Create these minimal pages (matching API above). They share the same backend; admin routes live at `/admin/*`.

```tsx
// client/src/pages/auth/ForgotPassword.tsx
import { useState } from 'react'

export default function ForgotPassword() {
  const [email, setEmail] = useState('')
  const [sent, setSent] = useState(false)

  async function submit(e: React.FormEvent) {
    e.preventDefault()
    await fetch('/api/auth/forgot-password', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email })
    })
    setSent(true)
  }

  if (sent) {
    return <div className="container py-16">
      <h1>Check your email</h1>
      <p>If an account exists, a reset link was sent.</p>
    </div>
  }

  return (
    <div className="container py-16">
      <h1>Forgot your password?</h1>
      <form onSubmit={submit} className="space-y-4 max-w-md">
        <input value={email} onChange={e=>setEmail(e.target.value)} placeholder="Email" className="input" />
        <button className="btn btn-primary">Send reset link</button>
      </form>
    </div>
  )
}
```

```tsx
// client/src/pages/auth/ResetPassword.tsx
import { useSearchParams } from 'react-router-dom'
import { useState } from 'react'

export default function ResetPassword() {
  const [params] = useSearchParams()
  const token = params.get('token') ?? ''
  const [password, setPassword] = useState('')
  const [done, setDone] = useState(false)

  async function submit(e: React.FormEvent) {
    e.preventDefault()
    await fetch('/api/auth/reset-password', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ token, password })
    })
    setDone(true)
  }

  if (!token) return <div className="container py-16">Invalid token</div>
  if (done) return <div className="container py-16">Password updated. You can sign in now.</div>

  return (
    <div className="container py-16">
      <h1>Set a new password</h1>
      <form onSubmit={submit} className="space-y-4 max-w-md">
        <input type="password" value={password} onChange={e=>setPassword(e.target.value)} placeholder="New password" className="input" />
        <button className="btn btn-primary">Update password</button>
      </form>
    </div>
  )
}
```
