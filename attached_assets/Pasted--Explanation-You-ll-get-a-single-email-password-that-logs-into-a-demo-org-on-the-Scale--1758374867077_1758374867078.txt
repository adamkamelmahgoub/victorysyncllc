// server/db/seedDemoScale.ts
import { db } from "./client";
import { organizations, users } from "./schema";
import { eq } from "drizzle-orm";
import bcrypt from "bcryptjs";
import { applyPlanToOrg } from "../billing/stripe";

async function run() {
  const orgId = "demo-scale";
  const email = "demo@victorysync.com";
  const password = "VictoryScale!23";

  // Ensure org exists
  const existing = await db.query.organizations.findFirst({ where: eq(organizations.id, orgId) });
  if (!existing) {
    await db.insert(organizations).values({
      id: orgId, name: "VictorySync Demo (Scale)", timezone: "America/New_York", planId: "scale",
    });
    await applyPlanToOrg(orgId, "scale"); // writes all Scale entitlements
    console.log("Created org with Scale plan:", orgId);
  }

  // Upsert user
  const hash = await bcrypt.hash(password, 10);
  // delete + insert keeps things idempotent for the demo
  await db.execute(`delete from users where email = $1`, [email]);
  await db.insert(users).values({
    id: "user-demo-scale",
    email,
    passwordHash: hash,
    orgId,
    role: "demo",
  });
  console.log("✅ Demo user ready:", { email, password });
  process.exit(0);
}
run().catch((e)=>{ console.error(e); process.exit(1); });

## 3) Auth middleware (JWT from cookie or header)

```ts
// server/middleware/auth.ts
import type { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";

export function auth(req: Request, _res: Response, next: NextFunction) {
  const bearer = req.headers.authorization?.startsWith("Bearer ")
    ? req.headers.authorization.split(" ")[1]
    : undefined;
  const token = (req as any).cookies?.vs_token || bearer;
  if (!token) { (req as any).auth = null; return next(); }

  try {
    const payload = jwt.verify(token, process.env.JWT_SECRET!) as any;
    (req as any).auth = { userId: payload.sub, orgId: payload.orgId, role: payload.role };
  } catch {
    (req as any).auth = null;
  }
  next();
}
```

## 4) Auth routes: **login** and **logout**

```ts
// server/routes/auth.ts
import { Router } from "express";
import cookie from "cookie";
import jwt from "jsonwebtoken";
import bcrypt from "bcryptjs";
import { db } from "../db/client";
import { users, organizations } from "../db/schema";
import { and, eq } from "drizzle-orm";

export const authRouter = Router();

authRouter.post("/login", async (req, res) => {
  const { email, password } = req.body as { email: string; password: string };
  if (!email || !password) return res.status(400).json({ error: "Email and password required" });

  const user = await db.query.users.findFirst({ where: eq(users.email, email) });
  if (!user) return res.status(401).json({ error: "Invalid credentials" });

  const ok = await bcrypt.compare(password, user.passwordHash);
  if (!ok) return res.status(401).json({ error: "Invalid credentials" });

  const org = await db.query.organizations.findFirst({ where: eq(organizations.id, user.orgId) });
  const token = jwt.sign({ sub: user.id, orgId: user.orgId, role: user.role }, process.env.JWT_SECRET!, { expiresIn: "24h" });

  res.setHeader("Set-Cookie", cookie.serialize("vs_token", token, {
    httpOnly: true, sameSite: "lax", path: "/",
    secure: process.env.NODE_ENV === "production",
    maxAge: 60 * 60 * 24,
  }));
  res.json({ ok: true, orgId: user.orgId, plan: org?.planId ?? null });
});

authRouter.post("/logout", async (_req, res) => {
  res.setHeader("Set-Cookie", cookie.serialize("vs_token", "", {
    httpOnly: true, sameSite: "lax", path: "/", secure: process.env.NODE_ENV === "production", maxAge: 0,
  }));
  res.json({ ok: true });
});

authRouter.get("/session", async (req, res) => {
  const a = (req as any).auth;
  res.json({ authenticated: !!a, orgId: a?.orgId ?? null, userId: a?.userId ?? null });
});
```

# Instructions

1. **Install deps (one time)**

   ```
   npm i bcryptjs jsonwebtoken cookie cookie-parser
   ```

2. **Seed the demo org + user**

   ```
   npx tsx server/db/seedDemoScale.ts
   ```

3. **Set secrets in Replit**

   * `JWT_SECRET` (any strong random string)
   * Your existing DB/Stripe/etc secrets remain as-is.

4. **Run the app**

   ```
   npm run dev
   ```

5. **Log in (from the browser or curl)**

   * Email: `demo@victorysync.com`
   * Password: `VictoryScale!23`

   Example (curl):

   ```
   curl -i -X POST http://localhost:5000/api/auth/login \
     -H "Content-Type: application/json" \
     -d '{"email":"demo@victorysync.com","password":"VictoryScale!23"}'
   ```

   The response sets the `vs_token` cookie. Your UI should now show all **Scale** features because entitlements exist for `demo-scale`.

6. **(Optional) Logout**

   ```
   POST /api/auth/logout
   ```
