# Explanation

That error means Postgres is still comparing a **text** org id to an **integer** somewhere. Your earlier attempt tried to add a FK from `users.org_id (integer)` → `organizations.id` while `organizations.id` is still **varchar**, so Postgres refused it.

Fix it permanently by:

1. **Normalizing `organizations.id` to an integer PK** (keep the old text in `id_legacy`).
2. **Adding the FKs** once types match.
3. **Always binding org ids as numbers in code** (tiny helper below).

Do this once and the `character varying = integer` problem is gone for good.

# Code

## 1) Paste this **single migration** into Neon’s SQL console and run

(Idempotent; it converts only if needed and skips nonexistent tables gracefully.)

```sql
BEGIN;

-- Ensure entitlements table has column "value" (your code reads it)
CREATE TABLE IF NOT EXISTS entitlements (
  id         SERIAL PRIMARY KEY,
  org_id     INTEGER NOT NULL,
  feature_id INTEGER NOT NULL,
  enabled    BOOLEAN NOT NULL DEFAULT TRUE
);
ALTER TABLE entitlements
  ADD COLUMN IF NOT EXISTS value TEXT;
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM information_schema.table_constraints
    WHERE table_name='entitlements' AND constraint_name='entitlements_org_feature_key'
  ) THEN
    ALTER TABLE entitlements
      ADD CONSTRAINT entitlements_org_feature_key UNIQUE (org_id, feature_id);
  END IF;
END $$;

-- If organizations.id is NOT integer/bigint, create a numeric PK and swap.
DO $$
DECLARE v_type text;
BEGIN
  SELECT data_type INTO v_type
  FROM information_schema.columns
  WHERE table_schema='public' AND table_name='organizations' AND column_name='id';

  IF v_type NOT IN ('integer','bigint') THEN
    -- 1) add numeric column
    IF NOT EXISTS (
      SELECT 1 FROM information_schema.columns
      WHERE table_schema='public' AND table_name='organizations' AND column_name='id_new'
    ) THEN
      EXECUTE 'ALTER TABLE organizations ADD COLUMN id_new BIGINT GENERATED BY DEFAULT AS IDENTITY';
    END IF;

    -- 2) keep numeric-looking legacy ids
    EXECUTE $Q$
      UPDATE organizations
      SET id_new = CASE WHEN id ~ '^[0-9]+$' THEN (id)::bigint ELSE id_new END
    $Q$;

    -- 3) make it NOT NULL
    EXECUTE 'ALTER TABLE organizations ALTER COLUMN id_new SET NOT NULL';

    -- 4) drop old PK (if any) and set new one
    IF EXISTS (
      SELECT 1 FROM information_schema.table_constraints
      WHERE table_schema='public' AND table_name='organizations' AND constraint_type='PRIMARY KEY'
    ) THEN
      EXECUTE 'ALTER TABLE organizations DROP CONSTRAINT organizations_pkey';
    END IF;
    EXECUTE 'ALTER TABLE organizations ADD CONSTRAINT organizations_pkey PRIMARY KEY (id_new)';

    -- 5) keep old text id in id_legacy, then rename new column to id
    IF NOT EXISTS (
      SELECT 1 FROM information_schema.columns
      WHERE table_schema='public' AND table_name='organizations' AND column_name='id_legacy'
    ) THEN
      EXECUTE 'ALTER TABLE organizations RENAME COLUMN id TO id_legacy';
    END IF;
    EXECUTE 'ALTER TABLE organizations RENAME COLUMN id_new TO id';
  END IF;
END $$;

-- Helpful columns / indexes
ALTER TABLE organizations
  ADD COLUMN IF NOT EXISTS name       VARCHAR(255),
  ADD COLUMN IF NOT EXISTS plan_id    VARCHAR(32),
  ADD COLUMN IF NOT EXISTS timezone   VARCHAR(64) DEFAULT 'America/New_York',
  ADD COLUMN IF NOT EXISTS created_at TIMESTAMP    DEFAULT NOW(),
  ADD COLUMN IF NOT EXISTS id_text    TEXT GENERATED ALWAYS AS (id::text) STORED;

DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_indexes WHERE schemaname='public' AND indexname='organizations_name_idx'
  ) THEN
    CREATE UNIQUE INDEX organizations_name_idx ON organizations (name);
  END IF;
END $$;

-- Foreign keys (now that organizations.id is integer)
DO $$
BEGIN
  BEGIN
    IF NOT EXISTS (
      SELECT 1 FROM information_schema.table_constraints
      WHERE table_name='users' AND constraint_name='users_org_fk'
    ) THEN
      EXECUTE 'ALTER TABLE users
               ADD CONSTRAINT users_org_fk
               FOREIGN KEY (org_id) REFERENCES organizations(id) ON DELETE SET NULL';
    END IF;
  EXCEPTION WHEN undefined_table THEN
    -- users table might not exist in your schema; skip
    NULL;
  END;

  IF NOT EXISTS (
    SELECT 1 FROM information_schema.table_constraints
    WHERE table_name='entitlements' AND constraint_name='entitlements_org_fk'
  ) THEN
    EXECUTE 'ALTER TABLE entitlements
             ADD CONSTRAINT entitlements_org_fk
             FOREIGN KEY (org_id) REFERENCES organizations(id) ON DELETE CASCADE';
  END IF;
END $$;

COMMIT;
```

> Re-run is safe; it only changes what’s missing.

## 2) Seed (only if you still need a demo)

(Use your current seeder if it already works. Otherwise, this one binds an **integer** org id.)

```js
// scripts/seedDemoScale.mjs
import { neon, neonConfig } from "@neondatabase/serverless";
import WebSocket from "ws";
import bcrypt from "bcryptjs";
neonConfig.webSocketConstructor = WebSocket;
const sql = neon(process.env.DATABASE_URL);

const EMAIL="demo@victorysync.com", PASS="VictoryScale123!", NAME="Demo Admin";
const ORG="VictorySync Demo (Scale)", ORG_TZ="America/New_York", PLAN="scale";

async function upsertOrgId() {
  const r = await sql/*sql*/`
    INSERT INTO organizations (name, timezone, plan_id)
    VALUES (${ORG}, ${ORG_TZ}, ${PLAN})
    ON CONFLICT (name) DO UPDATE
      SET plan_id=EXCLUDED.plan_id, timezone=EXCLUDED.timezone
    RETURNING id`;
  return Number(r[0].id); // guaranteed integer
}

async function main(){
  if(!process.env.DATABASE_URL) throw new Error("DATABASE_URL not set");
  if(!process.env.JWT_SECRET)   throw new Error("JWT_SECRET not set");

  const orgId = await upsertOrgId();

  const hash = await bcrypt.hash(PASS, 10);
  await sql/*sql*/`
    INSERT INTO public_users (name,email,password_hash,provider,provider_id,social_accounts,privileges)
    VALUES (${NAME}, lower(${EMAIL}), ${hash}, 'local', NULL, '[]'::jsonb, '{"role":"super_admin"}'::jsonb)
    ON CONFLICT (email) DO UPDATE
      SET name=EXCLUDED.name, password_hash=EXCLUDED.password_hash,
          provider='local', provider_id=NULL, social_accounts='[]'::jsonb,
          privileges='{"role":"super_admin"}'::jsonb, email=lower(${EMAIL})
  `;

  try{
    await sql/*sql*/`
      INSERT INTO users (org_id,name,email,password_hash,two_factor_enabled,is_super_admin,is_active)
      VALUES (${orgId}, ${NAME}, lower(${EMAIL}), ${hash}, false, true, true)
      ON CONFLICT (email) DO UPDATE
        SET org_id=${orgId}, name=EXCLUDED.name, password_hash=EXCLUDED.password_hash,
            two_factor_enabled=false, is_super_admin=true, is_active=true, email=lower(${EMAIL})
    `;
  }catch(e){
    if(!String(e?.message||e).includes('relation "users" does not exist')) throw e;
  }

  await sql/*sql*/`
    INSERT INTO entitlements (org_id, feature_id, enabled, value)
    SELECT ${orgId}, f.id, TRUE, NULL::text
    FROM features f
    WHERE NOT EXISTS (
      SELECT 1 FROM entitlements e
      WHERE e.org_id=${orgId} AND e.feature_id=f.id
    )
  `;

  console.log("✅ Demo seeded:", EMAIL, PASS, "orgId:", orgId);
}
main().catch(e=>{ console.error(e); process.exit(1); });
```

## 3) Code-side safety (parse orgId before every SQL and in JWT)

Create `server/db/id.ts`:

```ts
export function toIntId(value: unknown, field = "id"): number {
  const n = Number(String(value ?? "").trim());
  if (!Number.isInteger(n) || n <= 0) throw new Error(`Invalid ${field}: "${value}"`);
  return n;
}
```

Use it in login + any place you query by org:

```ts
import { toIntId } from "../db/id";
import jwt from "jsonwebtoken";
import { sql } from "../db/client";

// after verifying email/password:
const orgId = toIntId(user.org_id ?? org?.id ?? req.body?.org_id, "org_id");

const token = jwt.sign({ uid: user.id, orgId }, process.env.JWT_SECRET!, { expiresIn: "7d" });

// later queries:
await sql/*sql*/`SELECT ... FROM entitlements WHERE org_id = ${orgId}`;
await sql/*sql*/`SELECT ... FROM organizations WHERE id = ${orgId}`;
```
