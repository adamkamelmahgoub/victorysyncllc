### 3.2 Migration runner `server/db/runMigrations.ts`

Create:

```ts
// server/db/runMigrations.ts
import fs from 'fs';
import path from 'path';
import sql from './index';

const MIGRATIONS_DIR = path.join(process.cwd(), 'server', 'db', 'migrations');

async function ensureTable() {
  await sql`
    CREATE TABLE IF NOT EXISTS schema_migrations (
      id SERIAL PRIMARY KEY,
      name TEXT UNIQUE NOT NULL,
      applied_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    )
  `;
}

async function applied(): Promise<Set<string>> {
  const rows = await sql`SELECT name FROM schema_migrations`;
  return new Set(rows.map((r: any) => r.name));
}

async function applyMigration(name: string, sqlText: string) {
  await sql`BEGIN`;
  try {
    await sql(sqlText as any);
    await sql`INSERT INTO schema_migrations (name) VALUES (${name})`;
    await sql`COMMIT`;
    console.log(`✅ Applied ${name}`);
  } catch (err) {
    await sql`ROLLBACK`;
    console.error(`❌ Failed ${name}`, err);
    process.exit(1);
  }
}

async function main() {
  await ensureTable();
  const done = await applied();
  const files = fs.readdirSync(MIGRATIONS_DIR)
    .filter(f => f.endsWith('.sql'))
    .sort();
  for (const f of files) {
    if (!done.has(f)) {
      const text = fs.readFileSync(path.join(MIGRATIONS_DIR, f), 'utf8');
      await applyMigration(f, text);
    }
  }
  console.log('🎉 Migrations complete');
}

main();
```

-- users, orgs, roles, privileges, 2FA, payments, API keys, discount codes, calls

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TABLE IF NOT EXISTS organizations (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TYPE user_role AS ENUM ('admin', 'staff', 'customer');

CREATE TABLE IF NOT EXISTS users (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  email CITEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,
  role user_role NOT NULL DEFAULT 'customer',
  org_id UUID REFERENCES organizations(id) ON DELETE SET NULL,
  is_active BOOLEAN NOT NULL DEFAULT TRUE,
  privileges JSONB NOT NULL DEFAULT '[]'::jsonb, -- array of strings
  allowed_payment_methods JSONB NOT NULL DEFAULT '["card"]'::jsonb,
  twofa_enabled BOOLEAN NOT NULL DEFAULT FALSE,
  twofa_secret TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS access_codes (
  code TEXT PRIMARY KEY,
  description TEXT,
  max_uses INT,
  used INT NOT NULL DEFAULT 0,
  expires_at TIMESTAMPTZ
);

CREATE TABLE IF NOT EXISTS api_keys (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  org_id UUID REFERENCES organizations(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  token_hash TEXT NOT NULL,
  is_active BOOLEAN NOT NULL DEFAULT TRUE,
  scopes TEXT[] NOT NULL DEFAULT ARRAY[]::TEXT[],
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS integrations (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),

  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS user_integrations (

  integration_id UUID REFERENCES integrations(id) ON DELETE CASCADE,
  is_enabled BOOLEAN NOT NULL DEFAULT FALSE,
  settings JSONB NOT NULL DEFAULT '{}'::jsonb,
  PRIMARY KEY (user_id, integration_id)
);

CREATE TABLE IF NOT EXISTS packages (
  id SERIAL PRIMARY KEY,

## 5) Auth: access-code signup + 2FA

Add simple endpoints (merge into your existing `server/routes.ts` or `server/auth.ts`). If you have a router already, add these handlers; otherwise create `server/routes/auth.ts` and mount it.

```ts
// server/routes/auth.ts
import { Router } from 'express';
import sql from '../db';
import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { authenticator } from 'otplib';

const router = Router();

router.post('/signup-with-code', async (req, res) => {
  try {
    const { email, password, accessCode } = req.body;
    if (process.env.ACCESS_CODE_SIGNUP !== 'ENABLED') return res.status(403).json({message:'disabled'});
    const [code] = await sql`SELECT * FROM access_codes WHERE code=${accessCode}`;
    if (!code) return res.status(400).json({message:'invalid code'});
    if (code.expires_at && new Date(code.expires_at) < new Date()) return res.status(400).json({message:'expired'});
    if (code.max_uses && code.used >= code.max_uses) return res.status(400).json({message:'max uses reached'});

    const hash = await bcrypt.hash(password, 10);
    const [u] = await sql`
      INSERT INTO users (email, password_hash)
      VALUES (${email}, ${hash})
      ON CONFLICT (email) DO NOTHING
      RETURNING id,email,role,is_active,twofa_enabled
    `;
    if (!u) return res.status(409).json({message:'email exists'});

    await sql`UPDATE access_codes SET used=used+1 WHERE code=${accessCode}`;

    const token = jwt.sign({ sub: u.id, role: u.role }, process.env.JWT_SECRET!, { expiresIn: '12h' });
    res.json({ token, user: u });
  } catch (e) {
    console.error(e);
    res.status(500).json({message:'signup failed'});
  }
});

router.post('/login', async (req, res) => {
  const { email, password, otp } = req.body;
  const [u] = await sql`SELECT * FROM users WHERE email=${email}`;
  if (!u) return res.status(401).json({message:'bad creds'});
  const ok = await bcrypt.compare(password, u.password_hash);
  if (!ok) return res.status(401).json({message:'bad creds'});

  if (u.twofa_enabled) {
    if (!otp || !u.twofa_secret || !authenticator.check(otp, u.twofa_secret)) {
      return res.status(401).json({message:'otp required or invalid'});
    }
  }
  const token = jwt.sign({ sub: u.id, role: u.role }, process.env.JWT_SECRET!, { expiresIn: '12h' });
  res.json({ token, user: { id: u.id, email: u.email, role: u.role, twofa_enabled: u.twofa_enabled } });
});

router.post('/2fa/setup', async (req, res) => {
  const userId = (req as any).user?.sub; // assume auth middleware sets this
  if (!userId) return res.status(401).end();
  const secret = authenticator.generateSecret();
  const otpauth = authenticator.keyuri('VictorySync', 'VictorySync', secret);
  await sql`UPDATE users SET twofa_secret=${secret}, twofa_enabled=true WHERE id=${userId}`;
  res.json({ otpauth });
});

export default router;
```

Create file `client/src/admin/access-control/AccessControlCenter.tsx`:

```tsx
import React, { useEffect, useState } from 'react';

type User = {
  id: string;
  email: string;
  role: 'admin'|'staff'|'customer';
  is_active: boolean;
  privileges: string[];
  allowed_payment_methods: string[];
};

async function jfetch(url: string, opts: RequestInit = {}) {
  const r = await fetch(url, { headers: { 'Content-Type':'application/json' }, ...opts });
  if (!r.ok) throw new Error(await r.text());
  return r.json();
}

export default function AccessControlCenter() {
  const [users, setUsers] = useState<User[]>([]);
  const [loading, setLoading] = useState(true);

  const load = async () => {
    setLoading(true);
    const data = await jfetch('/api/admin/users');
    setUsers(data);
    setLoading(false);
  };

  useEffect(() => { load(); }, []);

  const toggleActive = async (u: User) => {
    await jfetch(`/api/admin/users/${u.id}/access`, {
      method:'POST',
      body: JSON.stringify({ isActive: !u.is_active })
    });
    load();
  };

  return (
    <div className="space-y-4">
      <h2 className="text-xl font-semibold">Access Control</h2>
      {loading ? <div>Loading…</div> : (
        <table className="w-full text-sm">
          <thead><tr>
            <th className="text-left">Email</th>
            <th>Role</th>
            <th>Active</th>
            <th>Payments</th>
            <th>Privileges</th>
            <th>Actions</th>
          </tr></thead>
          <tbody>
            {users.map(u => (
              <tr key={u.id} className="border-t">
                <td className="py-2">{u.email}</td>
                <td className="text-center">{u.role}</td>
                <td className="text-center">{u.is_active ? 'Yes' : 'No'}</td>
                <td className="text-center">{u.allowed_payment_methods.join(', ')}</td>
                <td className="text-center">{u.privileges.join(', ')}</td>
                <td className="text-center">
                  <button className="px-2 py-1 border rounded" onClick={()=>toggleActive(u)}>
                    {u.is_active ? 'Suspend' : 'Activate'}
                  </button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      )}
    </div>
  );
}
```
