## 1) Twilio IVR (plan-aware hours & priority)

```ts
// server/routes/ivr.ts
import { Router } from "express";
import twilio from "twilio";
import { db } from "../db/client";
import { organizations } from "../db/schema";
import { eq } from "drizzle-orm";
import { DateTime } from "luxon";

const router = Router();
const VoiceResponse = twilio.twiml.VoiceResponse;

function inBusinessHours(tz: string) {
  const l = DateTime.now().setZone(tz);
  return l.weekday <= 5 && l.hour >= 9 && l.hour < 17; // M–F 9–5
}
function inExtendedHours(tz: string) {
  const l = DateTime.now().setZone(tz);
  return l.hour >= 8 && l.hour < 22; // 8a–10p any day
}

router.post("/voice/:orgId", async (req, res) => {
  const { orgId } = req.params;
  const org = await db.query.organizations.findFirst({ where: eq(organizations.id, orgId) });
  if (!org) return res.status(404).send("org not found");

  const vr = new VoiceResponse();
  const { planId, timezone } = org;

  const enqueue = (queue: string, priority: number) => {
    const e = vr.enqueue({ waitUrl: "", action: "" }, queue);
    e.parameter({ name: "orgId", value: orgId });
    e.parameter({ name: "priority", value: String(priority) });
  };

  // Priority by plan
  const priority = planId === "scale" ? 100 : planId === "growth" ? 50 : 10;

  if (planId === "scale") {
    enqueue(`vs:${orgId}:primary`, priority);  // always 24/7
  } else if (planId === "growth") {
    if (inExtendedHours(timezone)) enqueue(`vs:${orgId}:primary`, priority);
    else vr.say("Our office is currently closed. Please leave a message."); // or go to after-hours
  } else {
    if (inBusinessHours(timezone)) enqueue(`vs:${orgId}:primary`, priority);
    else vr.say("We are closed. Please call back during business hours.");
  }

  res.type("text/xml").send(vr.toString());
});

export default router;
```

> Wire this route in `server/index.ts`:
> `app.use("/ivr", ivrRouter);` and point your Twilio number’s Voice webhook to `/ivr/voice/:orgId`.

## 2) HubSpot connector (basic + 2-way sync)

```ts
// server/connectors/hubspot.ts
import fetch from "node-fetch";

const HUBSPOT_BASE = "https://api.hubapi.com";
const headers = (token: string) => ({
  Authorization: `Bearer ${token}`,
  "Content-Type": "application/json",
});

export async function hsUpsertContact(token: string, email: string, props: Record<string,string|number>) {
  const r = await fetch(`${HUBSPOT_BASE}/crm/v3/objects/contacts/${encodeURIComponent(email)}?idProperty=email`, {
    method: "PATCH",
    headers: headers(token),
    body: JSON.stringify({ properties: props }),
  });
  if (!r.ok) throw new Error(`HubSpot upsert failed: ${r.status} ${await r.text()}`);
  return r.json();
}

export async function hsCreateTicket(token: string, subject: string, content: string) {
  const r = await fetch(`${HUBSPOT_BASE}/crm/v3/objects/tickets`, {
    method: "POST",
    headers: headers(token),
    body: JSON.stringify({ properties: { subject, content, hs_pipeline: "0", hs_pipeline_stage: "1" } }),
  });
  if (!r.ok) throw new Error(`HubSpot ticket failed: ${r.status} ${await r.text()}`);
  return r.json();
}
```

```ts
    .where(and(eq(entitlements.orgId, orgId), eq(entitlements.featureId, "email_support"))).limit(1);
```

## 4) Reports: Puppeteer renderer using template

```ts
// server/jobs/reports.ts
import cron from "node-cron";
import fs from "fs/promises";
import path from "path";
import puppeteer from "puppeteer";

function fill(template: string, data: Record<string, any>) {
  // very small token replacer supporting {{token}} and {{#each arr}}...{{/each}}
  let html = template;
  // arrays
  html = html.replace(/{{#each (\w+)}}([\s\S]*?){{\/each}}/g, (_, key, block) => {
    const arr = data[key] || [];
    return arr.map((row: any) => block.replace(/{{(\w+)}}/g, (_m: any, k: string) => row[k] ?? "")).join("");
  });
  // scalars
  html = html.replace(/{{(\w+)}}/g, (_m, k) => (data[k] ?? ""));
  return html;
}

export async function generateReportPDF(outFile: string, data: Record<string, any>) {
  const template = await fs.readFile(path.join(process.cwd(), "report_template.html"), "utf-8");
  const html = fill(template, data);

  const browser = await puppeteer.launch({ headless: "new" });
  const page = await browser.newPage();
  await page.setContent(html, { waitUntil: "networkidle0" });
  await page.pdf({ path: outFile, format: "A4", printBackground: true, margin: { top: "16mm", right: "16mm", bottom: "16mm", left: "16mm" } });
  await browser.close();
}
```

# Warnings

* **Secure webhooks** (Stripe/HQ, HubSpot) with HMAC or provider signatures.
* **Puppeteer memory**: use a separate worker in heavy usage or batch per org.
* **PII**: never log inbound messages in plaintext; mask emails/phones in logs.
* **Rate limits**: backoff on HubSpot and email providers; add retry queues.
