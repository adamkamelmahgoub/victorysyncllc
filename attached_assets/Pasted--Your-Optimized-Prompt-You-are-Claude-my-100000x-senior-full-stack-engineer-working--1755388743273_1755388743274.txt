# Code (edit exactly these files)

### 1) `client/vite.config.ts` — robust + Replit-safe HMR

Replace the whole export with this:

```ts
import { defineConfig } from "vite";
import react from "@vitejs/plugin-react";
import path from "path";

// Detect Replit container
const isReplit = !!process.env.REPL_ID || !!process.env.REPLIT_DB_URL;

// When running in Replit, HMR off (prevents WS flaps → page flicker).
// Locally, keep secure WSS for browser previews that are https.
const hmrConfig = isReplit
  ? false
  : {
      protocol: "wss",
      clientPort: 443,
      path: "/hmr",
    };

export default defineConfig({
  plugins: [react({ fastRefresh: true })],
  resolve: {
    alias: {
      "@": path.resolve(__dirname, "src"),
      "@shared": path.resolve(__dirname, "../shared"),
    },
  },
  server: {
    host: true,
    port: 5173,
    strictPort: true,
    cors: true,
    allowedHosts: true,
    hmr: hmrConfig,
    watch: {
      // FS polling is more stable in Replit containers
      usePolling: true,
      interval: 300,
      awaitWriteFinish: { stabilityThreshold: 500, pollInterval: 300 },
    },
  },
  build: { outDir: "dist" },
});
```

### 2) `server/index.ts` — keep sockets alive (already mostly done; add if missing)

Add **after** `const app = express();` and **after** `app.listen`:

```ts
app.set("trust proxy", 1);

const server = app.listen(PORT, "0.0.0.0", () => {
  console.log(`🎉 VictorySync running on http://0.0.0.0:${PORT} (${NODE_ENV})`);
});

server.keepAliveTimeout = 70_000;
server.headersTimeout   = 75_000;
server.on("connection", (socket) => socket.setKeepAlive(true, 60_000));
```

### 3) **Dev only**: ensure no Service Worker forces refresh loops

If you have any service worker registration, add this tiny dev-only guard.

Create (or append) **`client/src/dev-sw-unregister.ts`**:

```ts
// Unregister any SW in dev to avoid cache/reload loops in preview
if (import.meta.env.DEV && "serviceWorker" in navigator) {
  navigator.serviceWorker.getRegistrations().then((regs) => {
    regs.forEach((r) => r.unregister().catch(() => {}));
  });
}
```

Import it once at the very top of **`client/src/main.tsx`**:

```ts
import "./dev-sw-unregister";
// ...existing imports below
```

This does nothing in prod builds.
