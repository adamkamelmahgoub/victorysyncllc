  await sql/* sql */`

### 2) Stripe integration with safe fallback + plan application

**Fix duplicate export/name issues by using unique names.**

**`server/billing/stripe.ts`**

```ts
import type { PlanId } from './plans';
import { Plans } from './plans';
import { neon } from '@neondatabase/serverless';
import { sql as sharedSql, ensureEntitlementsValueColumn, setOrgPlan } from '../db/compat';

const SQL = sharedSql; // alias for clarity

// Apply plan + entitlements in a single transaction
export async function applyPlanToOrgTransactional(orgId: string, planId: PlanId) {
  await ensureEntitlementsValueColumn();

  await SQL/* sql */`BEGIN`;
  try {
    await setOrgPlan(orgId, planId);

    // Clear prior entitlements for idempotency
    await SQL/* sql */`
      DELETE FROM entitlements WHERE org_id = ${orgId}::text
    `;

    const features = Plans[planId].features;
    for (const feature of features) {
      await SQL/* sql */`
        INSERT INTO entitlements (org_id, feature_id, enabled, value)
        VALUES (${orgId}::text, ${feature}, true, 'enabled')
        ON CONFLICT (org_id, feature_id) DO UPDATE
          SET enabled = EXCLUDED.enabled, value = EXCLUDED.value
      `;
    }

    await SQL/* sql */`COMMIT`;
  } catch (e) {
    await SQL/* sql */`ROLLBACK`;
    throw e;
  }
}

// Stripe: only if keys exist; otherwise we use fallback
const STRIPE_SECRET = process.env.STRIPE_SECRET_KEY;
let stripe: any = null;
if (STRIPE_SECRET) {
  // Lazy import to avoid build issues if stripe isn't installed or keys missing
  // @ts-ignore
  stripe = (await import('stripe')).default(STRIPE_SECRET, { apiVersion: '2022-11-15' });
}

export async function createCheckoutSessionOrFallback(params: {
  orgId: string;
  planId: PlanId;
  successUrl: string;
  cancelUrl: string;
}) {
  const { orgId, planId, successUrl, cancelUrl } = params;

  if (!stripe) {
    // Dev fallback: apply immediately
    await applyPlanToOrgTransactional(orgId, planId);
    return { url: successUrl, mode: 'fallback' as const };
  }

  // Real Stripe Checkout
  const price = Plans[planId].price;
  const session = await stripe.checkout.sessions.create({
    mode: 'payment',
    success_url: successUrl,
    cancel_url: cancelUrl,
    metadata: { orgId, planId },
    line_items: [
      {
        price_data: {
          currency: 'usd',
          unit_amount: price * 100,
          product_data: { name: `${Plans[planId].name} Plan` }
        },
        quantity: 1
      }
    ]
  });

  return { url: session.url as string, mode: 'stripe' as const };
}

// Webhook handler: apply plan on completion
export async function handleStripeWebhook(rawBody: string, sig: string | undefined) {
  if (!stripe) return { ok: true }; // nothing to do in fallback mode

  const endpointSecret = process.env.STRIPE_WEBHOOK_SECRET;
  if (!endpointSecret) return { ok: true };

  let event;
  try {
    event = stripe.webhooks.constructEvent(rawBody, sig, endpointSecret);
  } catch (err) {
    return { ok: false, error: 'Invalid signature' };
  }

  if (event.type === 'checkout.session.completed') {
    const session = event.data.object as any;
    const orgId = session.metadata?.orgId as string;
    const planId = session.metadata?.planId as PlanId;
    if (orgId && planId) {
      await applyPlanToOrgTransactional(orgId, planId);
    }
  }
  return { ok: true };
}
```

### 3) Billing routes

**`server/routes/billing.ts`**

```ts
import { Router } from 'express';
import type { PlanId } from '../billing/plans';
import { createCheckoutSessionOrFallback, handleStripeWebhook } from '../billing/stripe';
import { getOrgById } from '../db/compat';

const router = Router();

// Require auth middleware expected to populate req.user with { org_id, email, ... }
function requireAuth(req: any, res: any, next: any) {
  if (!req.user?.org_id) return res.status(401).json({ error: 'Unauthorized' });
  next();
}

// POST /api/billing/checkout  { planId }
router.post('/checkout', requireAuth, async (req, res) => {
  const { planId } = req.body as { planId: PlanId };
  if (!planId) return res.status(400).json({ error: 'Missing planId' });

  const orgId = String(req.user.org_id);
  const successUrl = `${req.protocol}://${req.get('host')}/customer-dashboard?upgraded=${planId}`;
  const cancelUrl = `${req.protocol}://${req.get('host')}/pricing?cancelled=true`;

  try {
    // Make sure org exists
    const org = await getOrgById(orgId);
    if (!org) return res.status(400).json({ error: 'Org not found' });

    const { url, mode } = await createCheckoutSessionOrFallback({ orgId, planId, successUrl, cancelUrl });
    res.json({ url, mode });
  } catch (e: any) {
    res.status(500).json({ error: e.message ?? 'Failed to start checkout' });
  }
});

// Stripe webhook (raw body)
router.post('/webhook', expressRawBody, async (req: any, res: any) => {
  const sig = req.headers['stripe-signature'] as string | undefined;
  try {
    const result = await handleStripeWebhook(req.rawBody, sig);
    if (!result.ok) return res.status(400).send(result.error);
    res.json({ ok: true });
  } catch (e: any) {
    res.status(500).json({ error: e.message ?? 'Webhook error' });
  }
});

// Helper to get raw body
function expressRawBody(req: any, res: any, next: any) {
  let data = '';
  req.setEncoding('utf8');
  req.on('data', (chunk: string) => (data += chunk));
  req.on('end', () => {
    req.rawBody = data;
    next();
  });
}

export default router;
```
