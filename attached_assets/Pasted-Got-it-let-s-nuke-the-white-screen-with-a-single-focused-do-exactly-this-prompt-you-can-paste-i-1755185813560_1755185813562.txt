```ts
// server/index.ts
import 'dotenv/config';
import express from 'express';
import cookieParser from 'cookie-parser';
import cors from 'cors';
import path from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const app = express();
const PORT = Number(process.env.PORT || 5000);

// Allow cookies from Replit preview
app.use(cors({ origin: true, credentials: true }));
app.use(cookieParser());
app.use(express.json({ limit: '1mb' }));

// Lightweight health
app.get('/health', (_req, res) => res.status(200).send('ok'));

// ---------- API ----------
import apiRouter from './routes.js'; // keep existing export as default
app.use('/api', apiRouter);

// ---------- Vite / Static ----------
async function setupVite(app: express.Express) {
  const clientRoot = path.resolve(__dirname, '../client');

  if (process.env.NODE_ENV === 'development') {
    const { createServer: createViteServer } = await import('vite');
    const vite = await createViteServer({
      root: clientRoot,
      appType: 'custom',
      server: {
        middlewareMode: true,
        host: true,
        allowedHosts: true,
        strictPort: false,
        hmr: {
          protocol: 'wss',
          clientPort: 443,
          overlay: false, // never block rendering with an overlay
        },
      },
    });

    // Vite middlewares first
    app.use(vite.middlewares);

    // SPA fallback for any non-/api GET
    app.use(async (req, res, next) => {
      if (req.method !== 'GET' || req.path.startsWith('/api')) return next();
      try {
        const fs = await import('fs/promises');
        const htmlPath = path.join(clientRoot, 'index.html');
        const html = await fs.readFile(htmlPath, 'utf8');
        const transformed = await vite.transformIndexHtml(req.originalUrl, html);
        res.status(200).set('Content-Type', 'text/html').end(transformed);
      } catch (err) {
        vite.ssrFixStacktrace(err as Error);
        next(err);
      }
    });
  } else {
    const dist = path.resolve(clientRoot, 'dist');
    app.use(express.static(dist));
    app.get('*', (req, res) => {
      if (req.path.startsWith('/api')) return res.status(404).json({ message: 'Not found' });
      res.sendFile(path.join(dist, 'index.html'));
    });
  }
}

setupVite(app).then(() => {
  app.listen(PORT, () => console.log(`[express] serving on port ${PORT}`));
});
```

```tsx
import React from 'react';
type State = { hasError: boolean; error?: Error };
export default class ErrorBoundary extends React.Component<React.PropsWithChildren, State> {
  state: State = { hasError: false };
  static getDerivedStateFromError(error: Error) { return { hasError: true, error }; }
  componentDidCatch(error: Error, info: unknown) { console.error('UI Error:', error, info); }
  render() {
    if (this.state.hasError) {
      return <div style={{ padding: 24 }}><h2>Something went wrong.</h2><p>Open DevTools → Console for details.</p></div>;
    }
    return this.props.children;
  }
}
```

## 4) Fix queries that returned `Response` (must return parsed data)

**Create `client/src/lib/api.ts` and then refactor fetches to use it:**

```ts
export async function api<T>(method: string, url: string, body?: unknown): Promise<T> {
  const res = await fetch(`/api${url}`, {
    method,
    credentials: 'include',
    headers: { 'Content-Type': 'application/json' },
    body: body ? JSON.stringify(body) : undefined,
  });

  if (res.status === 401) {
    if (!location.pathname.startsWith('/login')) {
      sessionStorage.setItem('postLoginRedirect', location.pathname + location.search);
      location.assign('/login');
    }
    throw new Error('Unauthorized');
  }

  if (!res.ok) throw new Error(await res.text().catch(() => `HTTP ${res.status}`));
  const ct = res.headers.get('content-type') || '';
  return ct.includes('application/json') ? await res.json() as T : (undefined as T);
}
```

**Example fix (do this wherever you used `fetch` or returned `Response`):**

```ts

```

This prevents boot-time unhandled promise behavior from killing the render and stops `Unauthenticated` JSON from appearing on `/`.
