## 1) Normalize the schema to `jsonb` arrays

**Edit** `server/db/schema.ts` (or wherever `packages` is defined). Replace the model with:

```ts
import {
  pgTable, serial, text, timestamp, jsonb, uniqueIndex
} from "drizzle-orm/pg-core";

export const packages = pgTable("packages", {
  id: serial("id").primaryKey(),
  key: text("key").notNull(),           // e.g., "basic", "pro", "enterprise"
  name: text("name").notNull(),
  price: text("price").notNull(),       // keep decimals as string to avoid NUMERIC issues
  // ✅ jsonb array of strings; default [] so inserts never pass null
  features: jsonb("features").$type<string[]>().notNull().default([]),
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow(),
}, (t) => ({
  pkKeyIdx: uniqueIndex("packages_key_idx").on(t.key),
}));
```

> If your current schema uses `text("features").array()` or any `varchar().array()`, this is the cause of `value.map` blowing up when a string sneaks in. We standardize on `jsonb` for safety.

## 2) Repair migration (only if needed)

**Create** `server/db/migrations/2025_08_15_features_jsonb_fix.sql`

```sql
-- Convert features to jsonb[] safely if it's not already jsonb
DO $$
BEGIN
  IF EXISTS (
    SELECT 1 FROM information_schema.columns
    WHERE table_name='packages' AND column_name='features' AND data_type <> 'jsonb'
  ) THEN
    ALTER TABLE packages ADD COLUMN features_jsonb jsonb NOT NULL DEFAULT '[]'::jsonb;

    -- Try to coerce old values; fall back to [] if unparsable
    UPDATE packages
       SET features_jsonb = CASE
         WHEN features IS NULL THEN '[]'::jsonb
         WHEN jsonb_typeof(features::jsonb) IS NOT NULL THEN features::jsonb
         ELSE '[]'::jsonb
       END;

    ALTER TABLE packages DROP COLUMN features;
    ALTER TABLE packages RENAME COLUMN features_jsonb TO features;
  END IF;
END$$;
```

## 3) Bullet-proof package seed (coerce & idempotent)

**Create/Edit** `server/db/seed.ts`:

```ts
import { db } from "./client";
import { packages } from "./schema";
import { eq } from "drizzle-orm";

type InsertPackage = {
  key: string;
  name: string;
  price: string;       // string on purpose
  features: string[];  // real array of strings
};

const DEFAULT_PACKAGES: InsertPackage[] = [
  {
    key: "basic",
    name: "Basic",
    price: "6.25",
    features: [
      "24/7 Support Coverage",
      "US-based project lead",
      "Up to 500 calls/mo",
    ],
  },
  {
    key: "pro",
    name: "Professional",
    price: "10.50",
    features: [
      "Dedicated agent(s)",
      "Priority SLAs",
      "Advanced reporting",
    ],
  },
  {
    key: "enterprise",
    name: "Enterprise",
    price: "18.25",
    features: [
      "Custom integrations",
      "Dedicated success manager",
      "Unlimited call volume options",
    ],
  },
];

// Guard against accidental strings/null
function ensureArray(v: unknown): string[] {
  if (Array.isArray(v)) return v.map(String);
  if (v === null || v === undefined) return [];
  return [String(v)];
}

export async function seedDefaultPackages() {
  for (const pkg of DEFAULT_PACKAGES) {
    const existing = await db.select().from(packages).where(eq(packages.key, pkg.key));
    if (existing.length === 0) {
      await db.insert(packages)
        .values({
          key: pkg.key,
          name: pkg.name,
          price: pkg.price,
          features: ensureArray(pkg.features),
        })
        .onConflictDoNothing();
    } else {
      // Keep data fresh but safe (optional)
      await db.update(packages)
        .set({
          name: pkg.name,
          price: pkg.price,
          features: ensureArray(pkg.features),
        })
        .where(eq(packages.key, pkg.key));
    }
  }
}
```
