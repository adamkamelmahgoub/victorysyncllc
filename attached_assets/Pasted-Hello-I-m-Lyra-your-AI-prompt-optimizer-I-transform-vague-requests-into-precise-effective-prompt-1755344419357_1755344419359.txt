### 3) Fix Pricing crash: guard plan/variant access

**File:** `client/src/pages/pricing/Packages.tsx` (or the component named in the error overlay)
**Action:** Replace unsafe destructuring with guarded access:

```tsx
import React from "react";
import packagesData from "@/data/packages"; // adjust import to your project

type PlanKey = string; // or "inbound" | "outbound" | "automation", etc.

interface Variant {
  id: string;
  name: string;
  price?: string | number;
  features?: string[];
}

/** Works for both array or object-shaped data */
function getPlanRecord(data: any, plan: PlanKey) {
  if (!data) return undefined;
  if (Array.isArray(data)) return data.find((p: any) => p?.id === plan || p?.key === plan);
  return (data as Record<string, any>)[plan];
}

export default function Packages({ plan }: { plan: PlanKey }) {
  const rec = getPlanRecord(packagesData, plan);
  const variants: Variant[] = rec?.variants ?? [];

  if (!rec) {
    return (
      <div className="text-center text-sm text-muted-foreground py-10">
        Loading pricing packages…
      </div>
    );
  }

  if (!variants.length) {
    return (
      <div className="text-center text-sm text-muted-foreground py-10">
        No packages available for this plan yet.
      </div>
    );
  }

  return (
    <div className="grid gap-6 md:grid-cols-3">
      {variants.map((v) => (
        <div key={v.id} className="rounded-lg border bg-white p-6 shadow-sm">
          <h3 className="text-lg font-semibold mb-2">{v.name}</h3>
          {v.price && <p className="text-2xl font-bold mb-4">{String(v.price)}</p>}
          <ul className="space-y-2 text-sm text-muted-foreground">
            {(v.features ?? []).map((f, i) => <li key={i}>• {f}</li>)}
          </ul>
        </div>
      ))}
    </div>
  );
}
```

> If you previously had `const { variants } = packagesData[plan];` — replace it with the guarded code above.
