```ts
import fs from "fs";
import path from "path";

const DATA_DIR = path.resolve(process.cwd(), ".data");
if (!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR);

function load<T>(file: string, init: T): T {
  const p = path.join(DATA_DIR, file);
  if (!fs.existsSync(p)) { fs.writeFileSync(p, JSON.stringify(init, null, 2)); return init; }
  try { return JSON.parse(fs.readFileSync(p, "utf-8")) as T; } catch { return init; }
}
function save<T>(file: string, data: T) {
  const p = path.join(DATA_DIR, file);
  fs.writeFileSync(p, JSON.stringify(data, null, 2));
}

export type AccessCode = {
  code: string; email?: string; org?: string;
  role?: "superAdmin"|"admin"|"manager"|"agent"|"customer";
  maxUses: number; usedCount: number;
  expiresAt?: string; createdAt: string; isActive: boolean;
};
type AccessState = { codes: AccessCode[] };
const accessFile = "access-codes.json";
export const AccessStore = {
  list(): AccessCode[] { return load<AccessState>(accessFile,{codes:[]}).codes; },
  get(code: string) { return AccessStore.list().find(c=>c.code===code)||null; },
  upsert(ac: AccessCode) { const s=load<AccessState>(accessFile,{codes:[]}); const i=s.codes.findIndex(c=>c.code===ac.code); if(i>=0)s.codes[i]=ac; else s.codes.push(ac); save(accessFile,s); }
};

export type User = {
  id: string; email: string; role: "superAdmin"|"admin"|"manager"|"agent"|"customer";
  org?: string; passwordHash: string; twoFAEnabled?: boolean; createdAt: string;
};
type UserState = { users: User[] };
const userFile = "users.json";
export const UserStore = {
  list(): User[] { return load<UserState>(userFile,{users:[]}).users; },
  byEmail(email: string) { return UserStore.list().find(u=>u.email.toLowerCase()===email.toLowerCase())||null; },
  byId(id: string) { return UserStore.list().find(u=>u.id===id)||null; },
  upsert(u: User){ const s=load<UserState>(userFile,{users:[]}); const i=s.users.findIndex(x=>x.id===u.id||x.email.toLowerCase()===u.email.toLowerCase()); if(i>=0)s.users[i]=u; else s.users.push(u); save(userFile,s); }
};

export type ResetToken = { id: string; email: string; isAdmin: boolean; otp: string; createdAt: string; verifiedAt?: string; };
type ResetState = { tokens: ResetToken[] };
const resetFile = "password-resets.json";
export const ResetStore = {
  list(): ResetToken[] { return load<ResetState>(resetFile,{tokens:[]}).tokens; },
  get(id: string){ return ResetStore.list().find(t=>t.id===id)||null; },
  upsert(t: ResetToken){ const s=load<ResetState>(resetFile,{tokens:[]}); const i=s.tokens.findIndex(x=>x.id===t.id); if(i>=0)s.tokens[i]=t; else s.tokens.push(t); save(resetFile,s); }
};
```

```ts
import { Express, Request, Response } from "express";
import jwt from "jsonwebtoken";
import argon2 from "argon2";
import { z } from "zod";
import { nanoid } from "nanoid";
import { AccessStore, UserStore, ResetStore, User } from "../lib/store";

const JWT_SECRET = process.env.JWT_SECRET || "dev-secret";
const ADMIN_EMAIL = process.env.ADMIN_EMAIL || "admin@victorysync.com";
const ADMIN_PASSWORD = process.env.ADMIN_PASSWORD || "victorysync2025";
const ADMIN_VERIFICATION_PHRASE = process.env.ADMIN_VERIFICATION_PHRASE || "VICTORYSYNC-OWNER";

async function ensureAdmin() {
  if (!UserStore.byEmail(ADMIN_EMAIL)) {
    const passwordHash = await argon2.hash(ADMIN_PASSWORD);
    UserStore.upsert({
      id: "u_admin",
      email: ADMIN_EMAIL,
      role: "superAdmin",
      passwordHash,
      createdAt: new Date().toISOString()
    });
  }
}
function sign(u: Pick<User,"id"|"email"|"role">){ return jwt.sign(u, JWT_SECRET, { expiresIn:"12h" }); }
function err(res: Response, code: number, msg: string){ return res.status(code).json({ error: msg }); }

export function registerAuthRoutes(app: Express) {
  ensureAdmin().catch(()=>{});

  app.post("/api/auth/login", async (req, res)=>{
    try{
      const { email, password } = z.object({ email:z.string().email(), password:z.string().min(1)}).parse(req.body);
      const u = UserStore.byEmail(email); if(!u) return err(res,401,"Invalid email or password");
      const ok = await argon2.verify(u.passwordHash, password); if(!ok) return err(res,401,"Invalid email or password");
      return res.json({ token: sign({id:u.id,email:u.email,role:u.role}), user:{ id:u.id,email:u.email,role:u.role }});
    }catch(e:any){ return err(res,400,e?.message||"Login failed"); }
  });

  app.post("/api/auth/register", async (req,res)=>{
    try{
      const body = z.object({ email:z.string().email(), password:z.string().min(8), name:z.string().min(1), org:z.string().min(1).optional(), accessCode:z.string().min(4)}).parse(req.body);
      const ac = AccessStore.get(body.accessCode); if(!ac || !ac.isActive) return err(res,400,"Invalid access code");
      if(ac.email && ac.email.toLowerCase()!==body.email.toLowerCase()) return err(res,400,"Access code not issued for this email");
      if(ac.expiresAt && new Date(ac.expiresAt) < new Date()) return err(res,400,"Access code expired");
      if(ac.usedCount >= ac.maxUses) return err(res,400,"Access code usage limit reached");
      if(UserStore.byEmail(body.email)) return err(res,400,"User already exists");
      const passwordHash = await argon2.hash(body.password);
      const u: User = { id:`u_${nanoid(10)}`, email: body.email, role: ac.role||"customer", org: body.org||ac.org, passwordHash, createdAt:new Date().toISOString() };
      UserStore.upsert(u); ac.usedCount += 1; AccessStore.upsert(ac);
      return res.json({ token: sign({id:u.id,email:u.email,role:u.role}), user: { id:u.id,email:u.email,role:u.role,org:u.org }});
    }catch(e:any){ return err(res,400,e?.message||"Registration failed"); }
  });

  // Customer forgot
  app.post("/api/auth/forgot/start",(req,res)=>{
    try{
      const { email } = z.object({ email:z.string().email() }).parse(req.body);
      const u = UserStore.byEmail(email); if(!u) return err(res,404,"User not found");
      const otp = Math.floor(100000+Math.random()*900000).toString(); const id=nanoid(16);
      ResetStore.upsert({ id, email, isAdmin:false, otp, createdAt:new Date().toISOString() });
      console.log(`[RESET OTP] customer email=${email} otp=${otp} tokenId=${id}`);
      return res.json({ resetId:id });
    }catch(e:any){ return err(res,400,e?.message||"Failed"); }
  });
  app.post("/api/auth/forgot/complete", async (req,res)=>{
    try{
      const { resetId, otp, newPassword } = z.object({ resetId:z.string().min(8), otp:z.string().min(4), newPassword:z.string().min(8)}).parse(req.body);
      const tok = ResetStore.get(resetId); if(!tok || tok.isAdmin) return err(res,400,"Invalid reset token");
      if(tok.otp !== otp) return err(res,400,"Invalid OTP");
      const u = UserStore.byEmail(tok.email); if(!u) return err(res,404,"User not found");
      u.passwordHash = await argon2.hash(newPassword); UserStore.upsert(u); tok.verifiedAt = new Date().toISOString(); ResetStore.upsert(tok);
      return res.json({ ok:true });
    }catch(e:any){ return err(res,400,e?.message||"Failed"); }
  });

  // Admin forgot (OTP + phrase)
  app.post("/api/auth/admin/forgot/start",(req,res)=>{
    try{
      const { email } = z.object({ email:z.string().email() }).parse(req.body);
      const u = UserStore.byEmail(email); if(!u || (u.role!=="admin" && u.role!=="superAdmin")) return err(res,404,"Admin not found");
      const otp = Math.floor(100000+Math.random()*900000).toString(); const id=nanoid(16);
      ResetStore.upsert({ id, email, isAdmin:true, otp, createdAt:new Date().toISOString() });
      console.log(`[RESET OTP] admin email=${email} otp=${otp} tokenId=${id}`);
      return res.json({ resetId:id, challenge:"Provide OTP and verificationPhrase" });
    }catch(e:any){ return err(res,400,e?.message||"Failed"); }
  });
  app.post("/api/auth/admin/forgot/complete", async (req,res)=>{
    try{
      const { resetId, otp, verificationPhrase, newPassword } = z.object({ resetId:z.string().min(8), otp:z.string().min(4), verificationPhrase:z.string().min(6), newPassword:z.string().min(12)}).parse(req.body);
      const tok = ResetStore.get(resetId); if(!tok || !tok.isAdmin) return err(res,400,"Invalid reset token");
      if(tok.otp !== otp) return err(res,400,"Invalid OTP");
      if(verificationPhrase !== ADMIN_VERIFICATION_PHRASE) return err(res,403,"Verification failed");
      const u = UserStore.byEmail(tok.email); if(!u) return err(res,404,"Admin not found");
      u.passwordHash = await argon2.hash(newPassword); UserStore.upsert(u); tok.verifiedAt = new Date().toISOString(); ResetStore.upsert(tok);
      return res.json({ ok:true });
    }catch(e:any){ return err(res,400,e?.message||"Failed"); }
  });

  app.get("/api/auth/me",(req,res)=>{
    try{
      const auth = req.header("authorization")||""; const [,raw]=auth.split(" "); if(!raw) return err(res,401,"Missing token");
      const user = jwt.verify(raw, JWT_SECRET); return res.json({ user });
    }catch{ return err(res,401,"Invalid token"); }
  });
}
```

```ts
import { Express, Request, Response } from "express";
import jwt from "jsonwebtoken";
import { z } from "zod";
import { nanoid } from "nanoid";
import { UserStore, AccessStore } from "../lib/store";

const JWT_SECRET = process.env.JWT_SECRET || "dev-secret";
function requireAdmin(req: Request, res: Response): {id:string,email:string,role:string}|null {
  try {
    const auth = req.header("authorization")||""; const [,raw]=auth.split(" ");
    if(!raw) throw new Error("no token");
    const u = jwt.verify(raw, JWT_SECRET) as any;
    if(!u || (u.role!=="admin" && u.role!=="superAdmin")) throw new Error("forbidden");
    return u;
  } catch { res.status(403).json({error:"Forbidden"}); return null; }
}

export function registerAdminRoutes(app: Express){
  // Users
  app.get("/api/admin/users",(req,res)=>{
    if(!requireAdmin(req,res)) return;
    const users = UserStore.list().map(u=>({ id:u.id, email:u.email, role:u.role, org:u.org, createdAt:u.createdAt }));
    res.json({ users });
  });
  app.patch("/api/admin/users/:id",(req,res)=>{
    if(!requireAdmin(req,res)) return;
    const { role, org } = z.object({ role:z.enum(["superAdmin","admin","manager","agent","customer"]).optional(), org:z.string().optional() }).parse(req.body);
    const u = UserStore.byId(req.params.id); if(!u) return res.status(404).json({error:"Not found"});
    if(role) u.role = role; if(org!==undefined) u.org = org; UserStore.upsert(u);
    res.json({ ok:true, user:{ id:u.id,email:u.email,role:u.role,org:u.org }});
  });

  // Invite (creates access code scoped to email)
  app.post("/api/admin/invite",(req,res)=>{
    const me = requireAdmin(req,res); if(!me) return;
    const { email, role, org } = z.object({
      email: z.string().email(),
      role: z.enum(["superAdmin","admin","manager","agent","customer"]).default("customer"),
      org: z.string().optional()
    }).parse(req.body);
    const code = `INV-${nanoid(8).toUpperCase()}`;
    const rec = { code, email, role, org, maxUses:1, usedCount:0, isActive:true, createdAt:new Date().toISOString() } as any;
    AccessStore.upsert(rec);
    res.json({ ok:true, accessCode: rec });
  });
}
```

```ts
export type HttpMethod = "GET"|"POST"|"PATCH"|"DELETE";
export async function apiRequest<T>(url: string, opts: { method?: HttpMethod; body?: any; token?: string } = {}): Promise<T> {
  const res = await fetch(url, {
    method: opts.method || "GET",
    headers: {
      "Content-Type": "application/json",
      ...(opts.token ? { Authorization: `Bearer ${opts.token}` } : {})
    },
    body: opts.body ? JSON.stringify(opts.body) : undefined,
  });
  const data = await res.json().catch(()=> ({}));
  if (!res.ok) throw new Error(data?.error || `Request failed: ${res.status}`);
  return data as T;
}
```

**Create `client/src/admin/pages/AccessCodes.tsx`:**

```tsx
import { useEffect, useState } from "react";
import { apiRequest } from "@/lib/apiRequest";

type Code = { code:string; email?:string; org?:string; role?:string; maxUses:number; usedCount:number; isActive:boolean; createdAt:string; expiresAt?:string; };
export default function AccessCodes(){
  const token = localStorage.getItem("vs_token")||"";
  const [rows,setRows]=useState<Code[]>([]);
  const [edit,setEdit]=useState<Record<string,{maxUses:number;isActive:boolean}>>({});

  function load(){ apiRequest<{codes:Code[]}>("/api/access-codes",{ token }).then(d=>setRows(d.codes)); }
  useEffect(()=>{ load(); },[]);

  async function patch(code:string){
    const c = edit[code]; if(!c) return;
    await apiRequest("/api/access-codes/"+code,{ method:"PATCH", token, body:{ maxUses:c.maxUses, isActive:c.isActive }});
    setEdit(s=>{ const n={...s}; delete n[code]; return n; });
    load();
  }

  return (
    <div className="space-y-6">
      <h2 className="text-xl font-semibold">Access Codes</h2>
      <div className="border rounded-xl overflow-hidden">
        <table className="w-full text-sm">
          <thead className="bg-gray-50">
            <tr><th className="p-2 text-left">Code</th><th className="p-2">Email</th><th className="p-2">Role</th><th className="p-2">Max</th><th className="p-2">Used</th><th className="p-2">Active</th><th className="p-2">Actions</th></tr>
          </thead>
          <tbody>
            {rows.map(r=>{
              const e = edit[r.code] ?? { maxUses:r.maxUses, isActive:r.isActive };
              return (
                <tr key={r.code} className="border-t">
                  <td className="p-2 text-left">{r.code}</td>
                  <td className="p-2">{r.email||"—"}</td>
                  <td className="p-2">{r.role||"customer"}</td>
                  <td className="p-2"><input type="number" className="w-20 border rounded-md px-2 py-1" value={e.maxUses} onChange={ev=>setEdit(s=>({...s,[r.code]:{...e,maxUses:Number(ev.target.value)}}))} /></td>
                  <td className="p-2">{r.usedCount}</td>
                  <td className="p-2">
                    <input type="checkbox" checked={e.isActive} onChange={ev=>setEdit(s=>({...s,[r.code]:{...e,isActive:ev.target.checked}}))}/>
                  </td>
                  <td className="p-2">
                    <button onClick={()=>patch(r.code)} className="rounded-md px-3 py-1 text-white" style={{background:"linear-gradient(90deg,#6D28D9,#4F46E5,#2563EB)"}}>Save</button>
                  </td>
                </tr>
              );
            })}
          </tbody>
        </table>
      </div>
    </div>
  );
}
```
