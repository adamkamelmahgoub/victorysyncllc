#### `client/src/pages/auth/ForgotPassword.tsx`

```tsx
import { useState } from 'react';

type Props = { variant?: 'customer' | 'admin' };

export default function ForgotPassword({ variant = 'customer' }: Props) {
  const [email, setEmail] = useState('');
  const [sent, setSent] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    setError(null);
    try {
      const res = await fetch(`/api/auth/forgot-password`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, variant }),
      });
      if (!res.ok) {
        const j = await res.json().catch(() => ({}));
        throw new Error(j.message || 'Request failed');
      }
      setSent(true);
    } catch (err: any) {
      setError(err.message || 'Something went wrong');
    }
  }

  return (
    <div className="mx-auto max-w-md py-16">
      <h1 className="text-2xl font-semibold mb-2">
        Forgot password ({variant})
      </h1>
      <p className="text-sm text-gray-500 mb-6">
        Enter your email and we’ll send a reset link.
      </p>

      {sent ? (
        <div className="rounded-md bg-green-50 p-4 text-green-700">
          If an account exists for <b>{email}</b>, a reset link has been sent.
        </div>
      ) : (
        <form onSubmit={submit} className="space-y-4">
          <input
            className="w-full rounded border px-3 py-2"
            type="email"
            required
            value={email}
            onChange={e => setEmail(e.target.value)}
            placeholder="you@company.com"
          />
          {error && (
            <div className="rounded-md bg-red-50 p-2 text-red-700 text-sm">
              {error}
            </div>
          )}
          <button className="w-full rounded bg-[#6D28D9] px-4 py-2 text-white">
            Send reset link
          </button>
        </form>
      )}
    </div>
  );
}
```

#### `client/src/pages/auth/ResetPassword.tsx`

```tsx
import { useMemo, useState } from 'react';

export default function ResetPassword() {
  const params = useMemo(() => new URLSearchParams(window.location.search), []);
  const token = params.get('token') || '';
  const [password, setPassword] = useState('');
  const [done, setDone] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    setError(null);
    try {
      const res = await fetch(`/api/auth/reset-password`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ token, password }),
      });
      if (!res.ok) {
        const j = await res.json().catch(() => ({}));
        throw new Error(j.message || 'Reset failed');
      }
      setDone(true);
    } catch (err: any) {
      setError(err.message || 'Something went wrong');
    }
  }

  return (
    <div className="mx-auto max-w-md py-16">
      <h1 className="text-2xl font-semibold mb-2">Set a new password</h1>
      {!token && (
        <div className="rounded-md bg-yellow-50 p-3 text-yellow-800 mb-4">
          Missing reset token. Please use the link sent to your email.
        </div>
      )}
      {done ? (
        <div className="rounded-md bg-green-50 p-4 text-green-700">
          Your password has been reset. You can now sign in.
        </div>
      ) : (
        <form onSubmit={submit} className="space-y-4">
          <input
            className="w-full rounded border px-3 py-2"
            type="password"
            required
            minLength={8}
            value={password}
            onChange={e => setPassword(e.target.value)}
            placeholder="New strong password"
          />
          {error && (
            <div className="rounded-md bg-red-50 p-2 text-red-700 text-sm">
              {error}
            </div>
          )}
          <button
            disabled={!token}
            className="w-full rounded bg-[#6D28D9] px-4 py-2 text-white disabled:opacity-50"
          >
            Reset password
          </button>
        </form>
      )}
    </div>
  );
}
```
