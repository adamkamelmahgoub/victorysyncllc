```ts
import { query } from '../db';
import { randomBytes } from 'crypto';
import { sql } from '@neondatabase/serverless';

export async function issueAccessCode(adminId: string, data: {
  email: string; phone: string; full_name: string; org_name: string; ttlMinutes?: number;
}) {
  const code = randomBytes(6).toString('hex').toUpperCase(); // 12 chars
  const expires = new Date(Date.now() + (data.ttlMinutes ?? 120)*60*1000);
  await query`
    INSERT INTO access_codes (code, issued_to_email, issued_to_phone, issued_to_full_name, issued_to_org_name, expires_at, created_by)
    VALUES (${code}, ${data.email}, ${data.phone}, ${data.full_name}, ${data.org_name}, ${expires.toISOString()}, ${adminId})
  `;
  return code;
}

export async function consumeAccessCode(code: string, payload: {
  email: string; phone: string; full_name: string; org_name: string;
}) {
  const rows = await query<any>`
    SELECT * FROM access_codes WHERE code=${code} AND consumed=false AND expires_at>now() LIMIT 1
  `;
  const ac = rows[0];
  if (!ac) throw new Error('Invalid or expired access code');
  // verify strict match
  if (ac.issued_to_email !== payload.email ||
      ac.issued_to_phone !== payload.phone ||
      ac.issued_to_full_name !== payload.full_name ||
      ac.issued_to_org_name !== payload.org_name) {
    throw new Error('Access code does not match submitted identity');
  }
  return ac;
}

export async function markAccessCodeConsumed(code: string, userId: string) {
  await query`UPDATE access_codes SET consumed=true, consumed_by=${userId} WHERE code=${code}`;
}
```

```ts
import { query } from '../db';
import bcrypt from 'bcrypt';

export async function findUserByEmail(email: string) {
  const r = await query<any>`SELECT * FROM users WHERE email=${email} LIMIT 1`; return r[0];

export async function createOrgIfNeeded(name: string) {
  const existing = await query<any>`SELECT id FROM organizations WHERE name=${name} LIMIT 1`;
  if (existing[0]) return existing[0].id;
  const ins = await query<any>`INSERT INTO organizations (name) VALUES (${name}) RETURNING id`; 
  return ins[0].id;
}
export async function createUser(data: {
  org_id: string; email: string; phone: string; full_name: string; role: string; password: string;
}) {
  const hash = await bcrypt.hash(data.password, 12);
  const ins = await query<any>`
    INSERT INTO users (org_id,email,phone,full_name,role,password_hash)
    VALUES (${data.org_id}, ${data.email}, ${data.phone}, ${data.full_name}, ${data.role}, ${hash})
    RETURNING id, email, role, org_id, full_name, phone;
  `;
  // default permissions + payment policy
  await query`INSERT INTO user_permissions (user_id) VALUES (${ins[0].id})`;
  await query`INSERT INTO user_payment_policies (user_id) VALUES (${ins[0].id})`;
  return ins[0];
}
export async function verifyPassword(user: any, pwd: string) {
  return bcrypt.compare(pwd, user.password_hash);
}
export async function setVerificationFlags(userId: string, {email, phone}:{email?:boolean; phone?:boolean}) {

  if (phone !== undefined) await query`UPDATE users SET phone_verified=${phone} WHERE id=${userId}`;
}
```

**Create** `server/services/otp.ts`:

```ts
import { randomInt } from 'crypto';
import { sendEmail } from '../lib/email';
import { sendSms } from '../lib/sms';

const cache = new Map<string,{email:string;phone:string; codeEmail:string; codeSms:string; expires:number}>();
export function generateOtps(key:string, email:string, phone:string) {
  const codeEmail = String(randomInt(100000,999999));
  const codeSms   = String(randomInt(100000,999999));
  cache.set(key, { email, phone, codeEmail, codeSms, expires: Date.now()+10*60*1000 });
  return { codeEmail, codeSms };
}
export async function sendOtps(email:string, phone:string, ce:string, cs:string) {
  await sendEmail(email, 'Your VictorySync Email Code', `<p>Your code: <b>${ce}</b></p>`);
  await sendSms(phone, `Your VictorySync SMS code: ${cs}`);
}
export function verifyOtps(key:string, emailCode:string, smsCode:string) {
  const rec = cache.get(key); if (!rec) return false;
  if (Date.now()>rec.expires) { cache.delete(key); return false; }
  const ok = rec.codeEmail===emailCode && rec.codeSms===smsCode;
  if (ok) cache.delete(key);
  return ok;
}
```

```ts
import { verify } from '../lib/jwt';
import type { Request, Response, NextFunction } from 'express';

export function requireAuth(req: Request, res: Response, next: NextFunction) {
  const token = req.cookies?.vs_token;
  if (!token) return res.status(401).json({message:'Unauthenticated'});
  try { (req as any).user = verify(token); next(); }
  catch { return res.status(401).json({message:'Invalid session'}); }
}
export function requireRole(...roles: string[]) {
  return (req: any, res: any, next: any) => {
    if (!roles.includes(req.user?.role)) return res.status(403).json({message:'Forbidden'});
    next();
  };
}
```

## 7) Admin Portal (real data)

**Replace** `client/src/pages/admin-portal.tsx`:

```tsx
import React, { useEffect, useState } from 'react';
import { api } from '@/api/client';

type UserRow = {
  id:string; email:string; phone:string; full_name:string; role:string; is_active:boolean;
  email_verified:boolean; phone_verified:boolean; org_name:string|null;
  can_use_api:boolean; can_manage_org:boolean; can_view_billing:boolean; integrations:Record<string,boolean>;
  allow_card:boolean; allow_paypal:boolean; allow_wire:boolean;
};

export default function AdminPortal(){
  const [users,setUsers] = useState<UserRow[]>([]);
  const [loading,setLoading] = useState(true);
  const [error,setError] = useState('');

  async function load(){
    try{ setLoading(true); setUsers(await api('/admin/users')); }
    catch(e:any){ setError(e.message); } finally { setLoading(false); }
  }
  useEffect(()=>{ load(); },[]);

  async function save(u:UserRow){
    await api(`/admin/users/${u.id}/policy`,{
      method:'POST',
      body:JSON.stringify({
        can_use_api:u.can_use_api, can_manage_org:u.can_manage_org, can_view_billing:u.can_view_billing,
        integrations:u.integrations, allow_card:u.allow_card, allow_paypal:u.allow_paypal, allow_wire:u.allow_wire,
        is_active:u.is_active
      })
    });
  }

  return (
    <main className="max-w-6xl mx-auto p-6">
      <h1 className="text-2xl font-semibold mb-4">Admin Portal</h1>
      {loading && <p>Loading…</p>}
      {error && <p className="text-red-600">{error}</p>}
      {!loading && users.map(u=>(
        <div key={u.id} className="rounded border p-4 mb-3 bg-white">
          <div className="font-medium">{u.full_name} <span className="text-gray-500">({u.email})</span> — {u.org_name||'No org'}</div>
          <div className="grid grid-cols-2 md:grid-cols-4 gap-2 my-3">
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.can_use_api} onChange={e=>setUsers(prev=>prev.map(x=>x.id===u.id?{...x,can_use_api:e.target.checked}:x))}/> API</label>
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.can_manage_org} onChange={e=>setUsers(prev=>prev.map(x=>x.id===u.id?{...x,can_manage_org:e.target.checked}:x))}/> Manage Org</label>
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.can_view_billing} onChange={e=>setUsers(prev=>prev.map(x=>x.id===u.id?{...x,can_view_billing:e.target.checked}:x))}/> View Billing</label>
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.is_active} onChange={e=>setUsers(prev=>prev.map(x=>x.id===u.id?{...x,is_active:e.target.checked}:x))}/> Active</label>
          </div>
          <div className="grid grid-cols-2 md:grid-cols-3 gap-2 my-3">
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.allow_card} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,allow_card:e.target.checked}:x))}/> Card</label>
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.allow_paypal} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,allow_paypal:e.target.checked}:x))}/> PayPal</label>
            <label className="flex items-center gap-2"><input type="checkbox" checked={u.allow_wire} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,allow_wire:e.target.checked}:x))}/> Wire</label>
          </div>
          <div className="grid grid-cols-2 md:grid-cols-4 gap-2 my-3">
            {['hubspot','salesforce','zohocrm','five9','talkdesk','gohighlevel'].map(k=>(
              <label key={k} className="flex items-center gap-2">
                <input type="checkbox" checked={!!u.integrations?.[k]} onChange={e=>setUsers(p=>p.map(x=>x.id===u.id?{...x,integrations:{...x.integrations,[k]:e.target.checked}}:x))}/>{k}
              </label>
            ))}
          </div>
          <button className="btn btn-primary" onClick={()=>save(u)}>Save</button>
        </div>
      ))}
    </main>
  );
}
```
