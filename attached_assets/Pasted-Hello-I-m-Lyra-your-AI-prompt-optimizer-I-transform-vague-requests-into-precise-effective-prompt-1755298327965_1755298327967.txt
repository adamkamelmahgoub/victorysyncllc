### 1) Create safe navigation primitives (internal/external)

#### Add: `client/src/components/SmartLink.tsx`

```tsx
import { Link, LinkProps } from "react-router-dom";
import React from "react";

type Props = {
  to: string;
  children: React.ReactNode;
  className?: string;
  newTab?: boolean;
} & Omit<React.AnchorHTMLAttributes<HTMLAnchorElement>, "href">;

const isInternal = (to: string) =>
  /^\/(?!\/)/.test(to) || to.startsWith("#"); // "/about", "/pricing", "#section"

export default function SmartLink({ to, children, className, newTab, ...rest }: Props) {
  if (isInternal(to)) {
    return (
      <Link to={to} className={className} {...(rest as Omit<LinkProps, "to">)}>
        {children}
      </Link>
    );
  }
  return (
    <a
      href={to}
      className={className}
      target={newTab ? "_blank" : rest.target}
      rel={newTab ? "noopener noreferrer" : rest.rel}
      {...rest}
    >
      {children}
    </a>
  );
}
```

import { useNavigate } from "react-router-dom";
import React from "react";

type Props = React.ButtonHTMLAttributes<HTMLButtonElement> & {
  to: string;
  replace?: boolean;
};

export default function NavButton({ to, replace, children, ...rest }: Props) {
  const navigate = useNavigate();
  return (
    <button
      {...rest}
      onClick={(e) => {
        rest.onClick?.(e);
        if (e.defaultPrevented) return; // allow custom cancel
        navigate(to, { replace });
      }}
    >
      {children}
    </button>
  );
}

### 2) Router that **auto-registers all pages** (no manual route drift)

#### Replace: `client/src/router.tsx`

```tsx
import { lazy, Suspense } from "react";
import { Routes, Route, Navigate } from "react-router-dom";
import RootLayout from "@/layouts/RootLayout";
import NotFound from "@/pages/NotFound";

type Loader = () => Promise<{ default: React.ComponentType<any> }>;
const pageModules = import.meta.glob("./pages/**/*.tsx") as Record<string, Loader>;

function fileToPath(file: string): string {
  let p = file.replace("./pages", "").replace(/\.tsx$/, "");
  p = p.replace(/\/?Index$/i, "");
  if (/^\/Home$/i.test(p)) p = "";
  p = p.replace(/\[([^\]]+)\]/g, ":$1");   // [slug] -> :slug
  p = (p ? p : "/").toLowerCase();
  return p;
}

const autoRoutes = Object.entries(pageModules)
  .map(([file, loader]) => ({ path: fileToPath(file), Component: lazy(loader) }))
  .filter(r => r.path !== "/404" && r.path !== "/notfound");

export default function AppRouter() {
  return (
    <Suspense fallback={null}>
      <Routes>
        <Route element={<RootLayout />}>
          {autoRoutes.filter(r => r.path === "/").map(({ path, Component }) =>
            <Route key={path} index element={<Component />} />
          )}
          {autoRoutes.filter(r => r.path !== "/").map(({ path, Component }) =>
            <Route key={path} path={path.slice(1)} element={<Component />} />
          )}
          <Route path="404" element={<NotFound />} />
          <Route path="*" element={<Navigate to="/404" replace />} />
        </Route>
      </Routes>
    </Suspense>
  );
}
```
