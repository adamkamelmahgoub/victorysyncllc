```tsx
import { createContext, useContext, useEffect, useState, PropsWithChildren } from "react";

type User = { id?: string; email?: string; name?: string } | null;
type AuthCtx = {
  user: User; authenticated: boolean; loading: boolean; error?: string;
  refresh: () => void; login: () => Promise<void>; logout: () => Promise<void>;
};
const AuthContext = createContext<AuthCtx | undefined>(undefined);

export function AuthProvider({ children }: PropsWithChildren) {
  const [user, setUser] = useState<User>(null);
  const [authenticated, setAuthed] = useState(false);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | undefined>();

  async function load() {
    try {
      setLoading(true);
      const res = await fetch("/api/auth/check", { credentials: "include" });
      const data = (await res.json().catch(() => ({}))) as any;
      setUser(data?.user ?? null); setAuthed(!!data?.authenticated); setError(undefined);
    } catch (e: any) { setUser(null); setAuthed(false); setError(String(e?.message||e)); }
    finally { setLoading(false); }
  }
  useEffect(() => { load(); }, []);

  return (
    <AuthContext.Provider value={{ user, authenticated, loading, error, refresh: load, login: async()=>{}, logout: async()=>{} }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth(): AuthCtx {
  const ctx = useContext(AuthContext);
  if (!ctx) {
    console.warn("useAuth called without AuthProvider — returning safe defaults.");
    return { user:null, authenticated:false, loading:false, error:"AuthProvider missing", refresh:()=>{}, login:async()=>{}, logout:async()=>{} };
  }
  return ctx;
}
```

### 2) Safe Cart Context + Legacy Compat

`client/src/contexts/CartContext.tsx` (CREATE/REPLACE)

```tsx
import { createContext, useContext, useMemo, useState, PropsWithChildren } from "react";
export type CartItem = { id: string; name?: string; price?: number; qty: number };
export type CartCtx = { items: CartItem[]; count: number; subtotal: number; add: (i:Omit<CartItem,"qty">, qty?:number)=>void; remove:(id:string)=>void; clear:()=>void; };
const CartContext = createContext<CartCtx | undefined>(undefined);

export function CartProvider({ children }: PropsWithChildren) {
  const [items, setItems] = useState<CartItem[]>([]);
  const add: CartCtx["add"] = (i, qty=1)=>setItems(p=>{const k=p.findIndex(x=>x.id===i.id); if(k>-1){const c=[...p]; c[k]={...c[k], qty:c[k].qty+qty}; return c;} return [...p,{...i, qty}]});
  const remove: CartCtx["remove"] = id => setItems(p=>p.filter(x=>x.id!==id));
  const clear = ()=> setItems([]);
  const count = useMemo(()=>items.reduce((n,i)=>n+i.qty,0),[items]);
  const subtotal = useMemo(()=>items.reduce((s,i)=>s+(i.price??0)*i.qty,0),[items]);
  return <CartContext.Provider value={{ items, count, subtotal, add, remove, clear }}>{children}</CartContext.Provider>;
}
export function useCartContext(): CartCtx {
  const ctx = useContext(CartContext);
  if (!ctx) { console.warn("useCartContext without provider — no-op cart."); return { items:[], count:0, subtotal:0, add:()=>{}, remove:()=>{}, clear:()=>{} }; }
  return ctx;
}
```

```tsx
import { Link, LinkProps } from "react-router-dom"; import React from "react";
type Props = { to: string; children: React.ReactNode; className?: string; newTab?: boolean; } & Omit<React.AnchorHTMLAttributes<HTMLAnchorElement>,"href">;
const isInternal = (to:string)=> /^\/(?!\/)/.test(to) || to.startsWith("#");
export default function SmartLink({ to, children, className, newTab, ...rest }: Props) {
  return isInternal(to)
    ? <Link to={to} className={className} {...(rest as Omit<LinkProps,"to">)}>{children}</Link>
    : <a href={to} className={className} target={newTab?"_blank":rest.target} rel={newTab?"noopener noreferrer":rest.rel} {...rest}>{children}</a>;
}
```

### 4) File-system Router for all pages

`client/src/router.tsx` (CREATE/REPLACE)

```tsx
import { lazy, Suspense } from "react";
import { Routes, Route, Navigate } from "react-router-dom";
import RootLayout from "@/layouts/RootLayout";
import NotFound from "@/pages/NotFound";

type Loader = () => Promise<{ default: React.ComponentType<any> }>;
const pageModules = import.meta.glob("./pages/**/*.tsx") as Record<string, Loader>;
function fileToPath(file:string){ let p=file.replace("./pages","").replace(/\.tsx$/,""); p=p.replace(/\/?Index$/i,""); if(/^\/Home$/i.test(p)) p=""; p=p.replace(/\[([^\]]+)\]/g,":$1"); return (p? p: "/").toLowerCase(); }
const auto = Object.entries(pageModules).map(([f,l])=>({ path:fileToPath(f), C:lazy(l)})).filter(r=>!["/404","/notfound"].includes(r.path));

export default function AppRouter(){
  return (<Suspense fallback={null}>
    <Routes>
      <Route element={<RootLayout/>}>
        {auto.filter(r=>r.path==="/").map(({C},i)=><Route key={i} index element={<C/>}/>)}
        {auto.filter(r=>r.path!=="/").map(({path,C})=><Route key={path} path={path.slice(1)} element={<C/>}/>)}
        <Route path="404" element={<NotFound/>}/><Route path="*" element={<Navigate to="/404" replace/>}/>
      </Route>
    </Routes>
  </Suspense>);
}
```

```tsx
import { useState } from "react";
type VerifyPayload={accessCode:string; name:string; email:string; phone:string; orgNumber:string;};
export default function SignUp(){
  const [step,setStep]=useState<"verify"|"otp"|"done">("verify");
  const [form,setForm]=useState<VerifyPayload>({accessCode:"",name:"",email:"",phone:"",orgNumber:""});
  const [otp,setOtp]=useState(""); const [error,setError]=useState<string|null>(null); const [debugOtp,setDebugOtp]=useState<string|null>(null);
  const on=(k:keyof VerifyPayload)=> (e:any)=> setForm({...form,[k]:e.target.value});
  async function verify(){
    setError(null);
    const r=await fetch("/api/access/verify",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify(form)});
    const d=await r.json(); if(!r.ok||!d.ok){setError(d.error||"Verification failed");return;}
    const r2=await fetch("/api/access/request-otp",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({accessCode:form.accessCode,email:form.email,phone:form.phone})});
    const d2=await r2.json(); if(!r2.ok||!d2.ok){setError(d2.error||"Could not send OTP");return;}
    if(d2.devOtp) setDebugOtp(d2.devOtp); setStep("otp");
  }
  async function confirm(e:any){ e.preventDefault(); setError(null);
    const r=await fetch("/api/access/confirm-otp",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({accessCode:form.accessCode,email:form.email,otp})});
    const d=await r.json(); if(!r.ok||!d.ok){setError(d.error||"Invalid OTP");return;} setStep("done");
  }
  if(step==="done") return (<section className="max-w-md mx-auto p-6"><h1 className="text-2xl font-bold mb-2">Account Verified</h1><p className="mb-4">Your access code was verified and OTP confirmed.</p><a className="underline" href="/signin">Continue to Sign In</a></section>);
  return (<section className="max-w-md mx-auto p-6">
    <h1 className="text-2xl font-bold mb-4">Create Your Account</h1>
    {step==="verify" && (<form className="space-y-3" onSubmit={e=>{e.preventDefault(); verify();}}>
      <input className="w-full border p-2 rounded" placeholder="Access Code" value={form.accessCode} onChange={on("accessCode")} required />
      <input className="w-full border p-2 rounded" placeholder="Full Name" value={form.name} onChange={on("name")} required />
      <input className="w-full border p-2 rounded" placeholder="Work Email" value={form.email} onChange={on("email")} required />
      <input className="w-full border p-2 rounded" placeholder="Phone Number" value={form.phone} onChange={on("phone")} required />
      <input className="w-full border p-2 rounded" placeholder="Organization Number" value={form.orgNumber} onChange={on("orgNumber")} required />
      {error && <p className="text-red-600 text-sm">{error}</p>}
      <button className="w-full py-2 rounded bg-black text-white">Verify & Send OTP</button>
    </form>)}
    {step==="otp" && (<form className="space-y-3" onSubmit={confirm}>
      <input className="w-full border p-2 rounded" placeholder="Enter OTP" value={otp} onChange={e=>setOtp(e.target.value)} required />
      {debugOtp && <p className="text-xs text-gray-500">Dev OTP: {debugOtp}</p>}
      {error && <p className="text-red-600 text-sm">{error}</p>}
      <button className="w-full py-2 rounded bg-black text-white">Confirm OTP</button>
    </form>)}
  </section>);
}
```

```ts
import { Router } from "express";
import fs from "fs"; import path from "path";
export const api = Router();

api.get("/health", (_req,res)=>res.json({ok:true, ts:Date.now()}));
api.get("/auth/check", (_req,res)=>res.json({ ok:true, authenticated:false, user:null }));

type AccessCode={ code:string; orgNumber:string; orgName?:string; emailDomain?:string };
type OtpEntry={ key:string; otp:string; expiresAt:number };
const CODES_PATH=path.join(__dirname,"data","access-codes.json");
const OTPS_PATH=path.join(__dirname,"data","otps.json");
function readJson<T>(p:string,f:T):T{ try{return JSON.parse(fs.readFileSync(p,"utf8")) as T;}catch{return f;} }
function writeJson<T>(p:string,d:T){ fs.mkdirSync(path.dirname(p),{recursive:true}); fs.writeFileSync(p,JSON.stringify(d,null,2)); }
const genOtp=(n=6)=>Array.from({length:n},()=>Math.floor(Math.random()*10)).join("");

// Verify
api.post("/access/verify",(req,res)=>{
  const { accessCode, name, email, phone, orgNumber } = req.body??{};
  if(!accessCode||!name||!email||!phone||!orgNumber) return res.status(400).json({ok:false,error:"Missing required fields"});
  const codes=readJson<AccessCode[]>(CODES_PATH,[]); const found=codes.find(c=>c.code===accessCode);
  if(!found) return res.status(404).json({ok:false,error:"Invalid access code"});
  if(found.orgNumber && String(found.orgNumber)!==String(orgNumber)) return res.status(400).json({ok:false,error:"Organization number mismatch"});
  if(found.emailDomain && !String(email).toLowerCase().endsWith(`@${found.emailDomain.toLowerCase()}`)) return res.status(400).json({ok:false,error:`Email must end with @${found.emailDomain}`});
  res.json({ok:true, orgName:found.orgName??null});
});
// Request OTP
api.post("/access/request-otp",(req,res)=>{
  const { accessCode, email, phone } = req.body??{}; if(!accessCode||!email) return res.status(400).json({ok:false,error:"Missing accessCode or email"});
  const otps=readJson<OtpEntry[]>(OTPS_PATH,[]); const key=`${accessCode}|${String(email).toLowerCase()}`;
  const otp=genOtp(6); const expiresAt=Date.now()+5*60*1000;
  const i=otps.findIndex(o=>o.key===key); if(i>-1) otps[i]={key,otp,expiresAt}; else otps.push({key,otp,expiresAt}); writeJson(OTPS_PATH,otps);
  const devOtp=process.env.NODE_ENV!=="production"? otp: undefined; console.log(`[OTP] ${otp} -> ${email}${phone? " / "+phone: ""} for ${accessCode}`);
  res.json({ok:true, devOtp});
});
// Confirm OTP
api.post("/access/confirm-otp",(req,res)=>{
  const { accessCode, email, otp } = req.body??{}; if(!accessCode||!email||!otp) return res.status(400).json({ok:false,error:"Missing fields"});
  const otps=readJson<OtpEntry[]>(OTPS_PATH,[]); const key=`${accessCode}|${String(email).toLowerCase()}`; const e=otps.find(o=>o.key===key);
  if(!e) return res.status(400).json({ok:false,error:"No OTP pending"}); if(Date.now()>e.expiresAt) return res.status(400).json({ok:false,error:"OTP expired"});
  if(String(e.otp)!==String(otp)) return res.status(400).json({ok:false,error:"Invalid OTP"}); writeJson(OTPS_PATH,otps.filter(o=>o.key!==key)); res.json({ok:true});
});
```
