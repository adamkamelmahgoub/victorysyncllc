### `client/src/pages/Login.tsx`

```tsx
import { FormEvent, useState } from 'react'
import { useNavigate } from 'react-router-dom'
import { api } from '../lib/http'
import { useMe } from '../lib/auth'

export default function Login() {
  const [email, setEmail] = useState('demo@victorysync.com')
  const [password, setPassword] = useState('VictoryScale123!')
  const [error, setError] = useState<string | null>(null)
  const navigate = useNavigate()
  const { refetch } = useMe()

  const onSubmit = async (e: FormEvent) => {
    e.preventDefault()
    setError(null)
    const res = await api('/api/auth/login', {
      method: 'POST',
      body: JSON.stringify({ email, password }),
    })
    if (!res.ok) {
      setError('Invalid credentials')
      return
    }
    // Revalidate session & go to dashboard
    await refetch()
    navigate('/customer-dashboard')
  }

  return (
    <div style={{ maxWidth: 400, margin: '10vh auto', fontFamily: 'sans-serif' }}>
      <h2>Welcome Back</h2>
      <form onSubmit={onSubmit}>
        <label>Email</label>
        <input value={email} onChange={e => setEmail(e.target.value)} />
        <label>Password</label>
        <input type="password" value={password} onChange={e => setPassword(e.target.value)} />
        <button type="submit">Sign in</button>
      </form>
      {error && <p style={{ color: 'crimson' }}>{error}</p>}
    </div>
  )
}
```

```ts
  return fetch(path, {
```

### `client/src/lib/auth.ts`

```ts
import { useQuery } from '@tanstack/react-query'

export function useMe() {
  return useQuery({
    queryKey: ['me'],
    queryFn: async () => {
      const res = await fetch('/api/auth/me', { credentials: 'include' })
      if (!res.ok) return null
      return res.json() as Promise<{ id: number; email: string } | null>
    },
    retry: false,
  })
}
```

### `server/routes/auth.ts` (drop-in)

```ts
import type { Request, Response } from 'express'
import { Router } from 'express'
import jwt from 'jsonwebtoken'

// In-memory demo user for now — you can swap to DB users easily
const DEMO_USER = { id: 1, email: 'demo@victorysync.com', password: 'VictoryScale123!' }

const router = Router()

router.post('/login', (req: Request, res: Response) => {
  const { email, password } = req.body || {}
  if (email !== DEMO_USER.email || password !== DEMO_USER.password) {
    return res.status(401).json({ error: 'invalid' })
  }
  const token = jwt.sign({ sub: DEMO_USER.id }, process.env.JWT_SECRET!, { expiresIn: '1d' })
  res.cookie('vsid', token, {
    httpOnly: true,
    sameSite: 'lax',
    secure: false, // set true when behind https
    path: '/',
    maxAge: 24 * 60 * 60 * 1000,
  })
  return res.json({ ok: true })
})

router.post('/logout', (_req: Request, res: Response) => {
  res.clearCookie('vsid', { path: '/' })
  res.json({ ok: true })
})

router.get('/me', (req: Request, res: Response) => {
  const raw = req.cookies?.vsid
  if (!raw) return res.status(401).json({ ok: false })
  try {
    const decoded = jwt.verify(raw, process.env.JWT_SECRET!) as { sub: number }
    if (decoded?.sub !== DEMO_USER.id) return res.status(401).json({ ok: false })
    return res.json({ id: DEMO_USER.id, email: DEMO_USER.email })
  } catch {
    return res.status(401).json({ ok: false })
  }
})

export default router
```

### `server/index.ts` (drop-in)

```ts
import express from 'express'
import cookieParser from 'cookie-parser'
import path from 'node:path'
import fs from 'node:fs'

const app = express()
const PORT = Number(process.env.PORT || 5000)
const FRONTEND_URL = process.env.FRONTEND_URL || 'http://localhost:5173'
const CLIENT_DIST = path.resolve(process.cwd(), 'client', 'dist')
const INDEX_HTML = path.join(CLIENT_DIST, 'index.html')

app.use(express.json())
app.use(cookieParser())

// API routes
import apiRoutes from './routes/index'
app.use('/api', apiRoutes)

// Serve SPA if built, otherwise helpful message
app.get('*', (req, res, next) => {
  // Permit /api/* to be handled by API routes
  if (req.path.startsWith('/api/')) return next()

  if (fs.existsSync(INDEX_HTML)) {
    // Serve static assets
    app.use(express.static(CLIENT_DIST))
    return res.sendFile(INDEX_HTML)
  } else {
    res
      .status(200)
      .type('html')
      .send(
        `VictorySync API is running.<br/>Build the frontend to serve the SPA (see scripts below).<br/>Dev frontend available at <a href="${FRONTEND_URL}">${FRONTEND_URL}</a>.`
      )
  }
})

app.listen(PORT, () => {
  console.log(`No built SPA found at ${INDEX_HTML}. Serving API only. FRONTEND_URL=${FRONTEND_URL}`)
  console.log(`VictorySync API running on :${PORT}`)
})
```

> If you previously had `import routes from './routes'` without a default export, that caused `“does not provide an export named 'default'”`. The code above uses `export default` properly and mounts it at `/api`.

## 6) Fix the login loop (what we changed and why)

* All frontend `fetch` calls use `credentials: "include"` so the HTTP-only cookie is sent/received.
* The server sets cookie `vsid` with `sameSite: 'lax'`, `secure: false` for local (flip to `true` behind HTTPS), and `path: '/'`.
* `/api/auth/me` decodes JWT and returns `{ id, email }`. Frontend queries it on boot and routes accordingly.
* After successful `/api/auth/login`, we refetch `/api/auth/me` then `navigate('/customer-dashboard')`.

This removes the “authorized then back to login” behavior.

```ts
import type { Request, Response, NextFunction } from 'express'
import jwt from 'jsonwebtoken'

function requireAuth(req: Request, res: Response, next: NextFunction) {
  const token = req.cookies?.vsid
  if (!token) return res.status(401).json({ error: 'unauthorized' })
  try {
    jwt.verify(token, process.env.JWT_SECRET!)
    return next()
  } catch {
    return res.status(401).json({ error: 'unauthorized' })
  }
}

router.get('/account', requireAuth, (_req, res) => {

})
```
