## 2) Harden Vite config for Replit proxy (WSS + polling + stable watch)

**client/vite.config.ts** (replace the `export default defineConfig({ ... })` body with this):

```ts
import { defineConfig } from "vite";
import react from "@vitejs/plugin-react";
import path from "path";

export default defineConfig({
  plugins: [react({ fastRefresh: true })],
  resolve: {
    alias: {
      "@": path.resolve(__dirname, "src"),
      "@shared": path.resolve(__dirname, "../shared"),
    },
  },
  server: {
    host: true,
    port: 5173,
    strictPort: true,
    cors: true,
    allowedHosts: true,
    hmr: {
      // Replit preview is HTTPS → force secure WS to 443 to avoid mixed-content issues
      protocol: "wss",
      clientPort: 443,
      path: "/hmr", // fixed path improves some proxy setups
      overlay: true,
    },
    // Replit’s file system can miss chokidar native events → poll for stability
    watch: {
      usePolling: true,
      interval: 300,
      awaitWriteFinish: { stabilityThreshold: 500, pollInterval: 300 },
    },
  },
  build: { outDir: "dist" },
});
```

## 3) Trust proxy + long-lived sockets on the Node server

**server/index.ts**
Add these lines right **after** you create the `app` (before routes):

```ts
app.set("trust proxy", 1); // behind Replit proxy
```

When you start the server, capture the returned `http.Server` and extend timeouts:

```ts
const server = app.listen(PORT, "0.0.0.0", () => {
  console.log(`🎉 VictorySync running on http://0.0.0.0:${PORT} (${NODE_ENV})`);
  console.log(`Health: http://0.0.0.0:${PORT}/health`);
});

// Keep sockets alive long enough for proxy/HMR
server.keepAliveTimeout = 70_000;
server.headersTimeout   = 75_000;
server.on("connection", (socket) => socket.setKeepAlive(true, 60_000));
```

> These help prevent the proxy from tearing down idle sockets that Vite HMR reuses.
