## Step 3 — Fix the Pricing crash with a real CartProvider

### `client/src/context/cart.tsx`

```tsx
import React, { createContext, useContext, useState } from "react";

type CartItem = { id: string; qty: number };
type CartContextValue = {
  items: CartItem[];
  addItem: (id: string, qty?: number) => void;
  clear: () => void;
};

const CartContext = createContext<CartContextValue | undefined>(undefined);

export const CartProvider: React.FC<React.PropsWithChildren> = ({ children }) => {
  const [items, setItems] = useState<CartItem[]>([]);
  const addItem = (id: string, qty = 1) =>
    setItems(prev => {
      const i = prev.findIndex(x => x.id === id);
      if (i > -1) { const copy = [...prev]; copy[i] = { ...copy[i], qty: copy[i].qty + qty }; return copy; }
      return [...prev, { id, qty }];
    });
  const clear = () => setItems([]);
  return <CartContext.Provider value={{ items, addItem, clear }}>{children}</CartContext.Provider>;
};

export const useCartContext = () => {
  const ctx = useContext(CartContext);
  if (!ctx) throw new Error("useCartContext must be used within a CartProvider");
  return ctx;
};
```

## Step 5 — SPA fallback so refresh/deep links never 404

**Create a tiny Express server** that serves the built app and always falls back to `index.html`.

### `server/index.ts`

```ts
import express from "express";
import path from "path";

const app = express();
const PORT = process.env.PORT || 5173;
const dist = path.join(process.cwd(), "client", "dist");

app.use(express.static(dist));
app.get("*", (_, res) => res.sendFile(path.join(dist, "index.html")));

app.listen(PORT, () => console.log(`✅ Web server running on :${PORT}`));
```
