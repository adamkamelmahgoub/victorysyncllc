```ts
import express from "express"
import path from "node:path"
import cookieParser from "cookie-parser"
import cors from "cors"
import compression from "compression"
import helmet from "helmet"
import routes from "./routes"

const isDev = process.env.NODE_ENV !== "production"
const DEV_STATIC = process.env.DEV_STATIC === "1"
const PORT = Number(process.env.PORT || 5000)

async function main() {
  const app = express()
  app.disable("x-powered-by")
  app.use(helmet())
  app.use(cors({ origin: true, credentials: true }))
  app.use(cookieParser())
  app.use(express.json({ limit: "2mb" }))
  app.use(compression())

  app.get("/health", (_req, res) => res.json({ ok: true, env: process.env.NODE_ENV, time: new Date().toISOString() }))
  app.use("/api", routes)

  if (isDev && !DEV_STATIC) {
    const vite = await (await import("vite")).createServer({
      root: path.resolve(process.cwd(), "client"),
      server: { middlewareMode: true, hmr: { clientPort: 443 }, host: true, port: 5173, allowedHosts: true }
    })
    app.use(vite.middlewares)
    app.get(/^(?!\/api).*/, async (_req, res) => {
      const html = await vite.transformIndexHtml("/", "<!DOCTYPE html><html><head></head><body><div id='root'></div><script type='module' src='/src/main.tsx'></script></body></html>")
      res.status(200).set({ "Content-Type": "text/html" }).end(html)
    })
  } else {
    const dist = path.resolve(process.cwd(), "client/dist")
    app.use(express.static(dist))
    app.get(/^(?!\/api).*/, (_req, res) => res.sendFile(path.join(dist, "index.html")))
  }

  const server = app.listen(PORT, "0.0.0.0", () => {
    console.log(`🎉 VictorySync server running on :${PORT}`)
    console.log(`Health: http://0.0.0.0:${PORT}/health`)
  })
  process.on("SIGTERM", () => server.close(() => process.exit(0)))
  process.on("SIGINT", () => server.close(() => process.exit(0)))
}
main().catch((e) => { console.error("Fatal boot error:", e); process.exit(1) })
```

```ts
import { randomBytes, createCipheriv, createDecipheriv } from "node:crypto"
const keyB64 = process.env.DATA_ENCRYPTION_KEY || ""
const key = keyB64 ? Buffer.from(keyB64, "base64url") : Buffer.alloc(32)
const ivlen = 12
export function enc(data: string) { const iv=randomBytes(ivlen); const c=createCipheriv("aes-256-gcm",key,iv); const d=Buffer.concat([c.update(data,"utf8"),c.final()]); const tag=c.getAuthTag(); return Buffer.concat([iv,tag,d]).toString("base64url") }
export function dec(blob: string) { const b=Buffer.from(blob,"base64url"); const iv=b.subarray(0,ivlen); const tag=b.subarray(ivlen,ivlen+16); const d=b.subarray(ivlen+16); const c=createDecipheriv("aes-256-gcm",key,iv); c.setAuthTag(tag); return Buffer.concat([c.update(d),c.final()]).toString("utf8") }
```

```ts
import { Request, Response, NextFunction } from "express"
import { verifyJwt } from "../lib/jwt"

export function requireAuth(req: Request, res: Response, next: NextFunction) {
  const token = req.cookies?.token || (req.headers.authorization || "").replace(/^Bearer\s+/, "")
  if (!token) return res.status(401).json({ error: "Unauthorized" })
  try {
    const p = verifyJwt(token) as any
    ;(req as any).user = { id: Number(p.sub), orgId: p.orgId, role: p.role }
    return next()
  } catch {
    return res.status(401).json({ error: "Unauthorized" })
  }
}

export function authorize(required: string[] = []) {
  return (_req: Request, res: Response, next: NextFunction) => {
    const u = (_req as any).user
    if (!u) return res.status(401).json({ error: "Unauthorized" })
    if (u.role === "superAdmin") return next()
    // TODO: enforce from rolePermissions/userPermissions (Claude: implement via Drizzle)
    if (required.length === 0) return next()
    if (u.role === "admin") return next()
    return res.status(403).json({ error: "Forbidden" })
  }
}
```
