## 6) Express server with Vite middleware in dev, static in prod

Create/replace **`server/index.ts`**:

```ts
import express from 'express'
import path from 'path'
import cookieParser from 'cookie-parser'
import cors from 'cors'
import { fileURLToPath } from 'url'
import { createServer as createViteServer, ViteDevServer } from 'vite'
import fs from 'fs'

const __filename = fileURLToPath(import.meta.url)
const __dirname = path.dirname(__filename)

const isProd = process.env.NODE_ENV === 'production'
const PORT = Number(process.env.PORT || 5000)

async function createApp() {
  const app = express()

  app.use(express.json({ limit: '1mb' }))
  app.use(cookieParser())
  app.use(cors({ origin: true, credentials: true }))

  // Health check
  app.get('/health', (_req, res) => res.json({ ok: true }))

  // ===== API routes (keep your existing routers) =====
  // Import your existing routers here; ensure they are named exports and mounted.
  // Example:
  // import { authRouter } from './routes/auth.js'
  // import { billingRouter } from './routes/billing.js'
  // app.use('/api/auth', authRouter)
  // app.use('/api/billing', billingRouter)

  if (!isProd) {
    // ----- DEV: Vite middleware -----
    const vite: ViteDevServer = await createViteServer({
      server: { middlewareMode: true, hmr: { clientPort: 443 }, host: true },
      appType: 'custom',
    })

    app.use(vite.middlewares)

    // SPA fallback to client index.html via Vite
    app.use('*', async (req, res, next) => {
      if (req.originalUrl.startsWith('/api/')) return next()
      try {
        const url = req.originalUrl
        let template = fs.readFileSync(path.resolve(__dirname, '../client/index.html'), 'utf-8')
        template = await vite.transformIndexHtml(url, template)
        res.status(200).set({ 'Content-Type': 'text/html' }).end(template)
      } catch (e) {
        vite.ssrFixStacktrace(e as Error)
        next(e)
      }
    })
  } else {
    // ----- PROD: serve built static -----
    const distDir = path.resolve(__dirname, '../client/dist')
    const clientIndex = path.join(distDir, 'index.html')

    app.use(express.static(distDir, { index: false }))

    // SPA fallback for non-API routes
    app.get('*', (req, res, next) => {
      if (req.originalUrl.startsWith('/api/')) return next()
      res.sendFile(clientIndex)
    })
  }

  return app
}

createApp()
  .then((app) => {
    app.listen(PORT, () => {
      console.log(`[express] listening on :${PORT} (isProd=${isProd})`)
    })
  })
  .catch((err) => {
    console.error('Fatal boot error:', err)
    process.exit(1)
  })
```

> Notes:
>
> * Fixes the prior `__dirname is not defined in ES module scope`.
> * Prevents “Cannot GET /” by serving SPA fallback.
> * Prevents HMR disconnects by proxying Vite inside Express during dev.

## 7) Fix broken route imports/exports

Wherever we import routes like `import routes from "./routes"` and the file actually does **not** export a default, convert to this pattern:

* In each router file (e.g., `server/routes/auth.ts`), **export a named router**:

```ts
import { Router } from 'express'
export const authRouter = Router()

// define endpoints...
// authRouter.post('/login', ...
```

* In `server/index.ts`, **import named**:

```ts
import { authRouter } from './routes/auth.js'
app.use('/api/auth', authRouter)
```

> Remove any imports like `import routes from './routes'` unless `routes.ts` truly does `export default router`. This removes the “does not provide an export named 'default'” error.
