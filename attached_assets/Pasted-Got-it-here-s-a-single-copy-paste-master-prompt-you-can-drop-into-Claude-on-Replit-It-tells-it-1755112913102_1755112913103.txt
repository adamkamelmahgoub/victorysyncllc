## Step 2 — Client: always send cookies, handle 401s, and scroll to top

**Create/Replace** `client/src/lib/api.ts` — wraps `fetch` to always send cookies and JSON-decode:

```ts
// client/src/lib/api.ts
export async function api<T>(method: string, url: string, body?: unknown): Promise<T> {
  const res = await fetch(`/api${url}`, {
    method,
    credentials: 'include',
    headers: { 'Content-Type': 'application/json' },
    body: body ? JSON.stringify(body) : undefined,
  });

  if (res.status === 401) {
    // redirect to login on unauthorized
    if (!location.pathname.startsWith('/login')) {
      sessionStorage.setItem('postLoginRedirect', location.pathname + location.search);
      location.assign('/login');
    }
    throw new Error('Unauthorized');
  }

  if (!res.ok) {
    const text = await res.text().catch(() => '');
    throw new Error(text || `HTTP ${res.status}`);
  }

  // Return JSON if any, else undefined
  const ct = res.headers.get('content-type') || '';
  return ct.includes('application/json') ? (await res.json()) as T : (undefined as T);
}
```
