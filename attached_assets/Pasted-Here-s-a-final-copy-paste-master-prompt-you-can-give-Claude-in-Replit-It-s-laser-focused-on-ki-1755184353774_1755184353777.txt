import 'dotenv/config';
import express from 'express';
import cookieParser from 'cookie-parser';
import cors from 'cors';
import path from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const app = express();
const PORT = Number(process.env.PORT || 5000);

// CORS for Replit preview: reflect origin & allow cookies
app.use(cors({ origin: true, credentials: true }));
app.use(cookieParser());
app.use(express.json({ limit: '1mb' }));

app.get('/health', (_req, res) => res.status(200).send('ok'));

// ---- API routes under /api only ----
import apiRouter from './routes.js'; // keep your existing router default export to /api
app.use('/api', apiRouter);

// ---- Attach Vite (dev) or serve static (prod) ----
async function setupVite(app: express.Express) {
  if (process.env.NODE_ENV === 'development') {
    const { createServer: createViteServer } = await import('vite');
    const vite = await createViteServer({
      root: path.resolve(__dirname, '../client'),
      appType: 'custom',
      server: {
        middlewareMode: true,
        // Replit/HMR resilience — do NOT block rendering if WS drops
        hmr: {
          // Use the preview’s https origin; clientPort 443 works in Replit iframe
          protocol: 'wss',
          clientPort: 443,
          overlay: false,
        },
        allowedHosts: true,
        host: true,
        strictPort: false,
      },
    });

    // Vite middlewares first
    app.use(vite.middlewares);

    // SPA fallback for every non-/api GET
    app.use(async (req, res, next) => {
      if (req.method !== 'GET' || req.path.startsWith('/api')) return next();
      try {
        const fs = await import('fs/promises');
        const url = req.originalUrl;
        const html = await fs.readFile(
          path.resolve(__dirname, '../client/index.html'),
          'utf8'
        );
        const transformed = await vite.transformIndexHtml(url, html);
        res.status(200).set({ 'Content-Type': 'text/html' }).end(transformed);
      } catch (err) {
        vite.ssrFixStacktrace(err as Error);
        next(err);
      }
    });
  } else {
    const dist = path.resolve(__dirname, '../client/dist');
    app.use(express.static(dist));
    app.get('*', (req, res) => {
      if (req.path.startsWith('/api')) return res.status(404).json({ message: 'Not found' });
      res.sendFile(path.join(dist, 'index.html'));
    });
  }
}

setupVite(app).then(() => {
  app.listen(PORT, () => console.log(`[express] serving on port ${PORT}`));
});

## 4) Always send cookies and redirect to login on 401

**Create/ensure this wrapper and use it everywhere you call fetch (TanStack Query etc.):**

```ts
// client/src/lib/api.ts
export async function api<T>(method: string, url: string, body?: unknown): Promise<T> {
  const res = await fetch(`/api${url}`, {
    method,
    credentials: 'include',
    headers: { 'Content-Type': 'application/json' },
    body: body ? JSON.stringify(body) : undefined,
  });

  if (res.status === 401) {
    if (!location.pathname.startsWith('/login')) {
      sessionStorage.setItem('postLoginRedirect', location.pathname + location.search);
      location.assign('/login');
    }
    throw new Error('Unauthorized');
  }

  if (!res.ok) throw new Error(await res.text().catch(() => `HTTP ${res.status}`));
  const ct = res.headers.get('content-type') || '';
  return ct.includes('application/json') ? (await res.json() as T) : (undefined as T);
}
```
