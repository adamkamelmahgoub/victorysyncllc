# Explanation

You had three root causes for the “links not working / runtime” issues:

1. **Missing app providers** → your `<Pricing />` and `<Contact />` pages use `useCartContext` and React Query (`useMutation`) but the app wasn’t wrapped with `CartProvider` and `QueryClientProvider`.
2. **Internal links** → some navigations hit missing routes (e.g., `/resources/qa-scorecard.pdf`) or full page reloads; we fix this with proper SPA routes and a detail page.
3. **UI imports / case sensitivity** → `Footer`/`Navigation` path & case mismatches.

Below is a **surgical fix** that makes the whole site work cleanly, adds a simple but solid **sign-in → protected dashboard** flow, polishes the hero and transitions, and keeps your current structure. I’m not cloning Verint’s proprietary design; this mirrors the structure and polish and tailors the copy to VictorySync.

### 2) Minimal, safe Auth (client-side now; switches to your API automatically if present)

```tsx
// client/src/contexts/AuthContext.tsx — NEW
import { createContext, useContext, useEffect, useMemo, useState } from "react";

type User = { id: string; email: string };
type Ctx = { user: User | null; login: (email: string, password: string) => Promise<void>; logout: () => void; isAuthenticated: boolean; };
const AuthContext = createContext<Ctx | undefined>(undefined);
const KEY = "vs_auth_token";

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);

  useEffect(() => {
    const token = localStorage.getItem(KEY);
    if (token) try { setUser(JSON.parse(token)); } catch { localStorage.removeItem(KEY); }
  }, []);

  async function login(email: string, password: string) {
    // Try server first (if you later add /api/auth/login it will just work)
    try {
      const res = await fetch("/api/auth/login", {
        method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email, password })
      });
      if (res.ok) {
        const data = await res.json(); // expect { user }
        setUser(data.user); localStorage.setItem(KEY, JSON.stringify(data.user)); return;
      }
    } catch {}
    // Fallback local demo login
    const fake = { id: "dev", email };
    setUser(fake); localStorage.setItem(KEY, JSON.stringify(fake));
  }
  function logout() { setUser(null); localStorage.removeItem(KEY); }
  const value = useMemo(() => ({ user, login, logout, isAuthenticated: !!user }), [user]);
  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}
export function useAuth() { const v = useContext(AuthContext); if (!v) throw new Error("useAuth must be used within AuthProvider"); return v; }
```

```tsx
// client/src/ui/PrivateRoute.tsx — NEW
import { Navigate, useLocation } from "react-router-dom";
import { useAuth } from "../contexts/AuthContext";
export default function PrivateRoute({ children }: { children: React.ReactNode }) {
  const { isAuthenticated } = useAuth();
  const loc = useLocation();
  if (!isAuthenticated) return <Navigate to="/login" state={{ from: loc.pathname }} replace />;
  return <>{children}</>;
}
```

```tsx
import { FormEvent, useState } from "react";
import { useAuth } from "../contexts/AuthContext";
import { useLocation, useNavigate } from "react-router-dom";

export default function Login() {
  const { login } = useAuth();
  const [email, setEmail] = useState("demo@victorysync.com");
  const [password, setPassword] = useState("");
  const [loading, setLoading] = useState(false);
  const nav = useNavigate();

  return (

    </section>
```
