## 1) Routing primitives (guarantee SPA navigation)

**client/src/components/SmartLink.tsx**

```tsx
import React from "react";
import { Link, LinkProps } from "react-router-dom";

type Props = {
  to: string;
  children: React.ReactNode;
  className?: string;
  newTab?: boolean;
} & Omit<React.AnchorHTMLAttributes<HTMLAnchorElement>, "href">;

const isInternal = (to: string) => /^\/(?!\/)/.test(to) || to.startsWith("#");

export default function SmartLink({ to, children, className, newTab, ...rest }: Props) {
  return isInternal(to)
    ? <Link to={to} className={className} {...(rest as Omit<LinkProps,"to">)}>{children}</Link>
    : <a href={to} className={className} target={newTab ? "_blank" : rest.target}
         rel={newTab ? "noopener noreferrer" : rest.rel} {...rest}>{children}</a>;
}
```

**client/src/components/NavButton.tsx**

```tsx
import React from "react";
import { useNavigate } from "react-router-dom";

type Props = React.ButtonHTMLAttributes<HTMLButtonElement> & { to: string; replace?: boolean };

export default function NavButton({ to, replace, children, ...rest }: Props) {
  const navigate = useNavigate();
  return (
    <button
      {...rest}
      onClick={(e) => { rest.onClick?.(e); if (!e.defaultPrevented) navigate(to, { replace }); }}
    >
      {children}
    </button>
  );
}
```

> **Replace** any `<a href="/...">` (internal) or `window.location` usages with `SmartLink` / `NavButton`.

import { lazy, Suspense } from "react";
import { Routes, Route, Navigate } from "react-router-dom";

const Home     = lazy(() => import("@/pages/index"));
const Services = lazy(() => import("@/pages/services"));
const Pricing  = lazy(() => import("@/pages/pricing"));
const About    = lazy(() => import("@/pages/about"));
const Contact  = lazy(() => import("@/pages/contact"));
const SignIn   = lazy(() => import("@/pages/SignIn")); // create if missing
const NotFound = lazy(() => import("@/pages/NotFound"));

export default function AppRouter() {
  return (
    <Suspense fallback={null}>
      <Routes>
        <Route element={<RootLayout />}>
          <Route index element={<Home />} />
          <Route path="services" element={<Services />} />
          <Route path="pricing"  element={<Pricing />} />
          <Route path="about"    element={<About />} />
          <Route path="contact"  element={<Contact />} />
          <Route path="signin"   element={<SignIn />} />

          <Route path="*"        element={<Navigate to="/404" replace />} />
        </Route>
      </Routes>
    </Suspense>
  );
}

## 6) Pricing page: safe React Query usage

**client/src/pages/pricing.tsx** (or `pages/pricing/index.tsx`) — make the query safe and simple

```tsx
import { useQuery } from "@tanstack/react-query";
// import SmartLink/NavButton as needed
// import useCart if you need addItem()

export default function Pricing() {
  const { data: packages = [], isLoading, isError } = useQuery({
    queryKey: ["packages"],
    queryFn: async () => {
      const res = await fetch("/api/packages");
      if (!res.ok) throw new Error("Failed to load packages");
      return res.json();
    },
  });

  if (isLoading) return <div className="text-center py-10 text-sm text-muted-foreground">Loading pricing packages…</div>;
  if (isError)   return <div className="text-center py-10 text-sm text-red-600">Unable to load pricing at the moment.</div>;
  if (!packages.length) return <div className="text-center py-10 text-sm text-muted-foreground">No packages available yet.</div>;

  return (
    <div className="grid gap-6 md:grid-cols-3">
      {packages.map((p: any) => (
        <div key={p.id} className="rounded-lg border bg-white p-6 shadow-sm">
          <h3 className="text-lg font-semibold mb-2">{p.name}</h3>
          {p.price && <p className="text-2xl font-bold mb-4">{String(p.price)}</p>}
          <ul className="space-y-2 text-sm text-muted-foreground">
            {(p.features ?? []).map((f: string, i: number) => <li key={i}>• {f}</li>)}
          </ul>
        </div>
      ))}
    </div>
  );
}
```

> Avoid destructuring from a possibly-undefined record (`packages[plan].variants`). If you still need plan-based views, guard as `rec?.variants ?? []` as shown earlier.
