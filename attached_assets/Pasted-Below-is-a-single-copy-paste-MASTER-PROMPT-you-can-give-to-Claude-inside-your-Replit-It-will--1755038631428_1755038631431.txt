import { query } from '../db';
import { randomBytes } from 'crypto';

export async function issueAccessCode(adminId: string, data:{email:string;phone:string;full_name:string;org_name:string; ttlMinutes?:number;}) {
  const code = randomBytes(6).toString('hex').toUpperCase();
  const expires = new Date(Date.now() + (data.ttlMinutes ?? 120)*60*1000);
  await query`
    INSERT INTO access_codes (code, issued_to_email, issued_to_phone, issued_to_full_name, issued_to_org_name, expires_at, created_by)
    VALUES (${code}, ${data.email}, ${data.phone}, ${data.full_name}, ${data.org_name}, ${expires.toISOString()}, ${adminId})
  `;
  return code;
}
export async function validateAccessCode(code:string, payload:{email:string;phone:string;full_name:string;org_name:string;}) {
  const rows = await query<any>`SELECT * FROM access_codes WHERE code=${code} AND consumed=false AND expires_at>now() LIMIT 1`;
  const ac = rows[0]; if (!ac) throw new Error('Invalid or expired access code');
  if (ac.issued_to_email!==payload.email || ac.issued_to_phone!==payload.phone ||
      ac.issued_to_full_name!==payload.full_name || ac.issued_to_org_name!==payload.org_name) {
    throw new Error('Access code does not match submitted identity');
  }
  return ac;
}
export async function markConsumed(code:string, userId:string) {
  await query`UPDATE access_codes SET consumed=true, consumed_by=${userId} WHERE code=${code}`;
}

**server/services/otp.ts**

```ts
import { randomInt } from 'crypto';
import { sendEmail } from '../lib/email';
import { sendSms } from '../lib/sms';

const cache = new Map<string,{e:string;s:string;exp:number}>();

export function generate(email:string, phone:string) {
  const key = `${email}:${phone}`;
  const e = String(randomInt(100000,999999));
  const s = String(randomInt(100000,999999));
  cache.set(key, { e, s, exp: Date.now()+10*60*1000 });
  return { key, eCode:e, sCode:s };
}
export async function dispatch(email:string, phone:string, eCode:string, sCode:string) {
  await sendEmail(email, 'Your VictorySync Email Code', `<p>Your code: <b>${eCode}</b></p>`);
  await sendSms(phone, `VictorySync SMS code: ${sCode}`);
}
export function verifyPair(key:string, e:string, s:string) {
  const rec = cache.get(key); if (!rec) return false;
  if (Date.now()>rec.exp) { cache.delete(key); return false; }
  const ok = rec.e===e && rec.s===s; if (ok) cache.delete(key);
  return ok;
}
```

import { verify } from '../lib/jwt';
import type { Request, Response, NextFunction } from 'express';

export function requireAuth(req: Request, res: Response, next: NextFunction) {
  const token = (req as any).cookies?.vs_token;
  if (!token) return res.status(401).json({message:'Unauthenticated'});
  try { (req as any).user = verify(token); next(); } catch { return res.status(401).json({message:'Invalid session'}); }
}
export function requireRole(...roles: string[]) {
  return (req:any,res:any,next:any)=> roles.includes(req.user?.role) ? next() : res.status(403).json({message:'Forbidden'});
}

  async function start(e:React.FormEvent){ e.preventDefault(); setErr(''); setMsg('');
    const r = await api('/auth/register/start', { method:'POST', body:JSON.stringify(payload) });
    setToken(r.token); setStage('verify');
  }
  async function verify(e:React.FormEvent){ e.preventDefault(); setErr(''); setMsg('');
    const r = await api('/auth/register/verify',{ method:'POST', body:JSON.stringify({
      token, code:payload.code, email:payload.email, phone:payload.phone, full_name:payload.full_name, org_name:payload.org_name, password:payload.password,
      email_code:emailCode, sms_code:smsCode
    })});
    setMsg(`Welcome, ${r.user.full_name}!`); 
  }

  return (
    <main className="max-w-xl mx-auto p-6">
      <h1 className="text-2xl font-semibold mb-4">Create your VictorySync account</h1>
      {stage==='start' ? (
        <form onSubmit={start} className="space-y-3">
          <input className="input" placeholder="Access code" value={payload.code} onChange={e=>setPayload({...payload, code:e.target.value})} required />
          <input className="input" placeholder="Full name" value={payload.full_name} onChange={e=>setPayload({...payload, full_name:e.target.value})} required />
          <input className="input" placeholder="Organization" value={payload.org_name} onChange={e=>setPayload({...payload, org_name:e.target.value})} required />
          <input className="input" type="email" placeholder="Email" value={payload.email} onChange={e=>setPayload({...payload, email:e.target.value})} required />
          <input className="input" placeholder="Phone" value={payload.phone} onChange={e=>setPayload({...payload, phone:e.target.value})} required />
          <input className="input" type="password" placeholder="Password" value={payload.password} onChange={e=>setPayload({...payload, password:e.target.value})} required />
          <button className="btn btn-primary w-full">Send verification codes</button>
        </form>
      ) : (
        <form onSubmit={verify} className="space-y-3">
          <p className="text-sm text-gray-600">Enter the 6-digit codes we sent to your email and phone.</p>
          <input className="input" placeholder="Email code" value={emailCode} onChange={e=>setEmailCode(e.target.value)} required />
          <input className="input" placeholder="SMS code" value={smsCode} onChange={e=>setSmsCode(e.target.value)} required />
          <button className="btn btn-primary w-full">Complete registration</button>
        </form>
      )}
      {err && <p className="text-red-600 mt-3">{err}</p>}
      {msg && <p className="text-green-700 mt-3">{msg}</p>}
    </main>
  );
}
