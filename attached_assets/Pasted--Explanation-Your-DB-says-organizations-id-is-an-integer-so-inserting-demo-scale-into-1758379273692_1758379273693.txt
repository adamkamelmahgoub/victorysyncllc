# Explanation

Your DB says `organizations.id` is an **integer**, so inserting `"demo-scale"` into `id` blows up. Fix: **don’t set `id`**. Insert (or find) the org first, grab its numeric `id`, then use that `orgId` everywhere (user row + entitlements).

Below is a drop-in replacement for your reset script that:

* Finds the demo org by **name** (or creates it) and returns its **numeric id**.
* Upserts the demo user (`demo@victorysync.com` / `VictoryScale123!`) pointing to that id.
* Ensures **Scale** entitlements are materialized for that org (idempotent).

# Code

Replace `scripts/resetDemoUser.mjs` with:

```js
// scripts/resetDemoUser.mjs
// Create/repair a demo org on the SCALE plan and a demo user, using a numeric org id.

import { neon, neonConfig } from "@neondatabase/serverless";
import WebSocket from "ws";
import bcrypt from "bcryptjs";

neonConfig.webSocketConstructor = WebSocket;

const sql = neon(process.env.DATABASE_URL);

const EMAIL = "demo@victorysync.com";
const PASSWORD = "VictoryScale123!";
const ORG_NAME = "VictorySync Demo (Scale)";
const ORG_TZ = "America/New_York";
const PLAN = "scale";

async function getOrCreateDemoOrgId() {
  // 1) Try to find existing org by name
  const found = await sql/*sql*/`SELECT id, plan_id FROM organizations WHERE name = ${ORG_NAME} LIMIT 1`;
  if (found.length) {
    const orgId = found[0].id;
    // Ensure plan is 'scale'
    await sql/*sql*/`UPDATE organizations SET plan_id=${PLAN} WHERE id=${orgId} AND plan_id <> ${PLAN}`;
    return orgId;
  }
  // 2) Create a new org, let DB assign integer id
  const inserted = await sql/*sql*/`
    INSERT INTO organizations (name, timezone, plan_id)
    VALUES (${ORG_NAME}, ${ORG_TZ}, ${PLAN})
    RETURNING id
  `;
  return inserted[0].id;
}

async function main() {
  // Ensure unique index so email upsert works regardless of prior data
  await sql/*sql*/`CREATE UNIQUE INDEX IF NOT EXISTS users_email_idx ON users (email)`;

  const orgId = await getOrCreateDemoOrgId();

  // Upsert demo user (normalize email)
  const hash = await bcrypt.hash(PASSWORD, 10);
  await sql/*sql*/`
    INSERT INTO users (email, password_hash, org_id, role)
    VALUES (lower(${EMAIL}), ${hash}, ${orgId}, 'demo')
    ON CONFLICT (email) DO UPDATE
      SET password_hash = EXCLUDED.password_hash,
          org_id = ${orgId},
          role = 'demo',
          email = lower(${EMAIL})
  `;

  // Ensure all SCALE entitlements exist for this org (idempotent)
  await sql/*sql*/`
    INSERT INTO entitlements (org_id, feature_id, enabled, value)
    SELECT ${orgId} AS org_id, pf.feature_id, TRUE AS enabled, NULL::text AS value
    FROM plan_features pf
    WHERE pf.plan_id = ${PLAN}
      AND NOT EXISTS (
        SELECT 1 FROM entitlements e
        WHERE e.org_id = ${orgId} AND e.feature_id = pf.feature_id
      )
  `;

  console.log("✅ Demo org & user ready on SCALE");
  console.log("   orgId:", orgId);
  console.log("   email:", EMAIL);
  console.log("   pass :", PASSWORD);
}

main().catch((e) => {
  console.error("❌ resetDemoUser failed:", e?.message || e);
  process.exit(1);
});
```

# Warnings

* If your column names differ (`planId` vs `plan_id`, etc.), adjust the SQL accordingly.
* Normalize login email to `lowercase()` server-side so case/whitespace never causes “Invalid credentials”.
* Keep the demo creds gated in non-prod or rotate them regularly.
