import express from 'express';
import cors from 'cors';
import cookieParser from 'cookie-parser';
import routes from './routes';

const app = express();

// CRITICAL for Secure cookies behind Replit's HTTPS proxy
app.set('trust proxy', 1);

app.use(express.json());
app.use(cookieParser());

app.use(
  cors({
    origin: (origin, cb) => {
      if (!origin) return cb(null, true); // same-origin, curl, etc.
      // Allow Replit preview hosts in dev
      if (/^https:\/\/.*\.repl\.co$/.test(origin) || /^https:\/\/.*\.replit\.dev$/.test(origin)) {
        return cb(null, true);
      }
      // Allow an explicit front-end URL if provided
      if (process.env.FRONTEND_URL && origin === process.env.FRONTEND_URL) {
        return cb(null, true);
      }
      return cb(new Error(`CORS blocked: ${origin}`));
    },
    credentials: true,
  })
);

### 3) `server/routes/session.ts`

```ts
import { Router, type Request, type Response, type NextFunction } from 'express';
import jwt from 'jsonwebtoken';

export const sessionRouter = Router();

export type JWTPayload = {
  sub: string | number;
  orgId?: string | number;
  role?: string;
  email?: string;
};

export const JWT_SECRET = process.env.JWT_SECRET || 'dev-secret-please-set-JWT_SECRET';

export const baseCookieOptions = {
  httpOnly: true,
  sameSite: 'none' as const, // cross-site cookies (Replit preview)
  secure: true,              // HTTPS only; Replit preview is HTTPS
  path: '/',                 // send to all paths
  // Do NOT set "domain" in Replit previews
};

export function authRequired(req: Request, res: Response, next: NextFunction) {
  try {
    const fromCookie = req.cookies?.vs_session as string | undefined;
    const fromHeader = (req.get('authorization') || '').replace(/^Bearer\s+/i, '') || undefined;
    const token = fromCookie || fromHeader;
    if (!token) return res.status(401).json({ error: 'Missing token' });

    const decoded = jwt.verify(token, JWT_SECRET) as JWTPayload;
    (req as any).user = decoded;
    return next();
  } catch {
    return res.status(401).json({ error: 'Invalid token' });
  }
}

export function meHandler(req: Request, res: Response) {
  return res.json({ user: (req as any).user ?? null });
}

// /api/session/me — canonical "who am I?" endpoint
sessionRouter.get('/me', authRequired, meHandler);

// /api/session/logout — clears cookie using the SAME options used to set it
sessionRouter.post('/logout', (req, res) => {
  res.clearCookie('vs_session', baseCookieOptions);
  return res.json({ ok: true });
});

// Dev-only helper (optional): see cookies server-side
// sessionRouter.get('/debug/cookies', (req, res) => res.json({ cookies: req.cookies, hasAuth: !!req.cookies?.vs_session }));
```

### 4) `server/routes/auth.ts`

import { Router } from 'express';
import jwt from 'jsonwebtoken';
import { authRequired, baseCookieOptions, JWT_SECRET, meHandler } from './session';

export const authRouter = Router();

// POST /api/auth/login — demo login; swap for real DB lookup later
authRouter.post('/login', async (req, res) => {
  const { email, password } = (req.body || {}) as { email?: string; password?: string };

  const valid =
    (email === 'demo@victorysync.com' && password === 'VictoryScale123!') ||
    (email === 'admin@victorysync.com' && password === 'VictoryScale123!');

  if (!valid) return res.status(401).json({ error: 'Invalid credentials' });

  const payload = {
    sub: 1,
    email,
    orgId: 'demo-scale',
    role: 'super_admin',
  };

  const token = jwt.sign(payload, JWT_SECRET, { expiresIn: '7d' });

  res.cookie('vs_session', token, {
    ...baseCookieOptions,
    maxAge: 7 * 24 * 60 * 60 * 1000, // 7 days
  });

  return res.json({ ok: true });
});

// GET /api/auth/me — alias to /api/session/me so either path works
authRouter.get('/me', authRequired, meHandler);

### 5) `client/src/lib/api.ts`

```ts
export async function api<T = any>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`/api${path}`, {
    credentials: 'include', // REQUIRED so cookie is sent/received
    headers: { 'Content-Type': 'application/json' },
    ...init,
  });
  if (!res.ok) {
    const text = await res.text().catch(() => '');
    throw new Error(text || `Request failed: ${res.status}`);
  }
  return res.json();
}
```

### 8) `client/src/pages/Login.tsx`

```tsx
import { FormEvent, useState } from 'react';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { useNavigate } from 'react-router-dom';
import { api } from '../lib/api';

export default function Login() {
  const [email, setEmail] = useState('demo@victorysync.com');
  const [password, setPassword] = useState('VictoryScale123!');
  const [err, setErr] = useState<string | null>(null);
  const qc = useQueryClient();
  const navigate = useNavigate();

  const loginMut = useMutation({
    mutationFn: async (body: { email: string; password: string }) =>
      api('/auth/login', { method: 'POST', body: JSON.stringify(body) }),
    onSuccess: async () => {
      await qc.invalidateQueries({ queryKey: ['/api/session/me'] });
      navigate('/customer-dashboard');
    },
    onError: (e: any) => setErr(e?.message || 'Login failed'),
  });

  const onSubmit = (e: FormEvent) => {
    e.preventDefault();
    setErr(null);
    loginMut.mutate({ email, password });
  };

  return (
    <div className="min-h-screen flex items-center justify-center p-6">
      <form onSubmit={onSubmit} className="w-full max-w-sm space-y-4 border rounded-xl p-6 shadow-sm">
        <h1 className="text-xl font-semibold">Sign in</h1>
        {err && <div className="text-red-600 text-sm">{err}</div>}
        <div className="space-y-1">
          <label className="text-sm">Email</label>
          <input
            className="w-full border rounded-md px-3 py-2"
            type="email"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            autoComplete="username"
            required
          />
        </div>
        <div className="space-y-1">
          <label className="text-sm">Password</label>
          <input
            className="w-full border rounded-md px-3 py-2"
            type="password"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            autoComplete="current-password"
            required
          />
        </div>
        <button
          type="submit"
          disabled={loginMut.isPending}
          className="w-full bg-indigo-600 text-white rounded-md py-2 font-medium hover:bg-indigo-700 disabled:opacity-70"
        >
          {loginMut.isPending ? 'Signing in…' : 'Sign in'}
        </button>
      </form>
    </div>
  );
}
```

  <React.StrictMode>
    <QueryClientProvider client={qc}>

        <AppRoutes />

### Why this works

* `app.set('trust proxy', 1)` ensures Express allows `Secure` cookies through the Replit reverse proxy.
* CORS is configured for Replit preview origins with `credentials: true`.
* Cookie is `httpOnly`, `Secure`, `SameSite=None` so the browser stores/sends it in cross-site iframes/previews.
* Frontend always requests with `credentials: 'include'` and uses a **consistent** `/api/session/me` path and query key, preventing the “undefined query data” + auto-logout loop.
