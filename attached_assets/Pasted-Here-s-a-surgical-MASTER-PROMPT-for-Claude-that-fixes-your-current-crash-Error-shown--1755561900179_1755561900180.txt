import express from 'express';
import path from 'path';
import cookieParser from 'cookie-parser';
import cors from 'cors';
import morgan from 'morgan';

import { fileURLToPath } from 'url'; // harmless in CJS build; keeps TS happy if needed
// IMPORTANT: routes must export a named registerRoutes(app) (see next section)
import { registerRoutes } from './routes';
import { ensureSuperAdmin } from './seed/superAdmin';

// In CJS build, __dirname is available. If bundler ever flips to ESM, fallback is below.
const __filename = typeof __dirname === 'undefined'
  ? fileURLToPath(import.meta.url)
  : '';
const __dirnameSafe = typeof __dirname === 'undefined'
  ? path.dirname(__filename)
  : __dirname;

const PORT = Number(process.env.PORT) || 5000;

async function bootstrap() {
  const app = express();

  // middlewares
  app.use(cors({ origin: true, credentials: true }));
  app.use(express.json({ limit: '10mb' }));
  app.use(cookieParser());
  app.use(morgan('tiny'));

  // seed super admin once (no top-level await)
  try {
    await ensureSuperAdmin();
  } catch (err) {
    console.error('[seed] ensureSuperAdmin failed:', err);
  }

  // start BullMQ scheduler only when REDIS_URL provided (no top-level await)
  if (process.env.REDIS_URL) {
    try {
      const { QueueScheduler } = await import('bullmq');
      // v4 no longer needs QueueScheduler for simple flows, but safe to init:
      // @ts-ignore
      new QueueScheduler('sync', { connection: { url: process.env.REDIS_URL! } });
      console.log('[bulmq] QueueScheduler started');
    } catch (err) {
      console.warn('[bulmq] QueueScheduler not started:', err?.message || err);
    }
  } else {
    console.log('[bulmq] QueueScheduler not started (no REDIS_URL)');
  }

  // API routes
  registerRoutes(app);

  // Serve client build in dev/prod as needed
  const distDir = path.resolve(__dirnameSafe, '../client/dist');
  app.use(express.static(distDir));

  // SPA fallback (never intercept /api)
  app.use((req, res, next) => {
    if (req.path.startsWith('/api')) return next();
    res.sendFile(path.join(distDir, 'index.html'));
  });

  app.listen(PORT, () => {
    console.log(`✅ API listening on :${PORT}`);
  });
