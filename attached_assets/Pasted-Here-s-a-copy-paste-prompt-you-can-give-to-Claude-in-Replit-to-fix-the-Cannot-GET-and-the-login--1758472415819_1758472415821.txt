```ts

  if (req.path.startsWith('/api')) return res.status(404).end();
```

### 3) Fix login loop: cookies/JWT & `/api/auth/me`

* Use **httpOnly** cookie for the access token (or a signed session cookie).
* SameSite should be `'lax'` (or `'strict'`) in Replit Preview so the cookie isn’t dropped.
* Set `secure: false` (Replit Preview is http).
* Ensure `/api/auth/me` reads the token from the cookie and returns `{ id, email, role, orgId, entitlements }`.

**Edit `server/routes/auth.ts`** (or wherever your auth lives):

```ts
import { Router } from 'express';
import jwt from 'jsonwebtoken';

const router = Router();
const JWT_SECRET = process.env.JWT_SECRET ?? 'dev-secret-change-me';
const COOKIE_NAME = 'vs_token';

function signToken(payload: object) {
  return jwt.sign(payload, JWT_SECRET, { expiresIn: '7d' });
}

function readToken(req: any) {
  const raw = req.cookies?.[COOKIE_NAME];
  if (!raw) return null;
  try {
    return jwt.verify(raw, JWT_SECRET);
  } catch {
    return null;
  }
}

// POST /api/auth/login { email, password }
router.post('/auth/login', async (req, res) => {
  const { email, password } = req.body ?? {};
  // TODO: replace with real user lookup + password verify
  const user = await fakeVerifyUser(email, password); // implement properly
  if (!user) return res.status(401).json({ message: 'Invalid credentials' });

  const token = signToken({
    sub: user.id,
    email: user.email,
    role: user.role,
    orgId: user.orgId,
  });

  res.cookie(COOKIE_NAME, token, {
    httpOnly: true,
    sameSite: 'lax',    // important to avoid cross-site drops in Preview
    secure: false,      // true only behind HTTPS
    path: '/',
    maxAge: 1000 * 60 * 60 * 24 * 7,
  });
  res.json({ ok: true });
});

// GET /api/auth/me
router.get('/auth/me', async (req, res) => {
  const claims = readToken(req);
  if (!claims) return res.status(401).json({ message: 'Unauthorized' });

  // TODO: fetch user and entitlements from DB using claims.sub/orgId
  const user = await fakeLoadUserWithEntitlements(claims.sub);
  res.json(user);
});

// POST /api/auth/logout
router.post('/auth/logout', (_req, res) => {
  res.clearCookie(COOKIE_NAME, { path: '/' });
  res.json({ ok: true });
});

export { router };
```

> Replace `fakeVerifyUser` / `fakeLoadUserWithEntitlements` with your real DB queries. The key is: token in httpOnly cookie + `/auth/me` uses that cookie.

### 4) Client login flow & guarding routes

**Edit `client/src/pages/login.tsx`** (example):

```tsx
const loginMutation = useMutation({
  mutationFn: async (form: { email: string; password: string }) => {
    const res = await fetch('/api/auth/login', {
      method: 'POST',
      credentials: 'include',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(form),
    });
    if (!res.ok) throw new Error('Login failed');
    return res.json();
  },
  onSuccess: () => {
    // refetch me, then navigate to dashboard
    queryClient.invalidateQueries({ queryKey: ['/api/auth/me'] });
    navigate('/customer-dashboard');
  },
});
```

**Bootstrap current user on app load**:

```tsx
// client/src/hooks/useMe.ts
import { useQuery } from '@tanstack/react-query';

export function useMe() {
  return useQuery({
    queryKey: ['/api/auth/me'],
    queryFn: async () => {
      const r = await fetch('/api/auth/me', { credentials: 'include' });
      if (r.status === 401) return null;
      if (!r.ok) throw new Error('me failed');
      return r.json();
    },
    staleTime: 5 * 60 * 1000,
  });
}
```

**Route guard**:

```tsx
// client/src/AppRoutes.tsx
function PrivateRoute({ children }: { children: React.ReactNode }) {
  const { data: me, isLoading } = useMe();
  if (isLoading) return <Spinner/>;
  if (!me) return <Navigate to="/login" replace />;
  return <>{children}</>;
}
```
