Got it — here’s a clean, **drop-in animation + hero upgrade** for every page using your lavender→pink palette. No packages, no config changes. Everything is pure React + CSS (IntersectionObserver), so it runs perfectly on Replit.

# Explanation

* Adds a reusable `<Reveal>` component (fade/slide-in on scroll).
* Enhances `HeroBanner` with a built-in animated “console” illustration (so every page has a hero image with depth, even without external assets).
* Extends your theme CSS with animation utilities and soft float/blur effects.
* Wraps all cards/sections in `<Reveal>` and turns on the hero panel across **all pages**.

# Code

### 1) Scroll-reveal utility (new)

**`src/components/Reveal.tsx`**

```tsx
import React, { useEffect, useRef } from "react";

type Variant = "up" | "fade" | "left" | "right";
type Tag = keyof JSX.IntrinsicElements;

export default function Reveal({
  as,
  children,
  className = "",
  variant = "up",
  delay = 0
}: {
  as?: Tag;
  children: React.ReactNode;
  className?: string;
  variant?: Variant;
  delay?: number; // 0..5 -> 0..300ms extra delay
}) {
  const Tag = (as || "div") as any;
  const ref = useRef<HTMLElement | null>(null);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const io = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          el.classList.add("is-visible");
          io.unobserve(el);
        }
      },
      { rootMargin: "0px 0px -10% 0px" }
    );
    io.observe(el);
    return () => io.disconnect();
  }, []);

  const delayClass =
    delay > 0 ? ` reveal-delay-${Math.min(5, Math.max(0, delay))}` : "";
  return (
    <Tag
      ref={ref as any}
      className={`reveal reveal-${variant}${delayClass} ${className}`}
    >
      {children}
    </Tag>
  );
}
```

**`src/components/HeroBanner.tsx`**

```tsx
import React from "react";
import "../styles/victorysync-theme.css";

type CTA = { label: string; to: string; variant?: "primary" | "outline" };

export default function HeroBanner({
  eyebrow,
  title,
  accent,
  subtitle,
  ctas = [],
  rightPanel = true,
  image,
  imageAlt
}: {
  eyebrow?: string;
  title: string;
  accent?: string;
  subtitle?: string;
  ctas?: CTA[];
  rightPanel?: boolean;
  image?: string;     // optional external/public image; if missing, we render an animated panel
  imageAlt?: string;
}) {
  return (
    <section className="hero-wrap">
      <div className="hero-bg" aria-hidden />
      <div className="mx-auto max-w-7xl px-4 py-16 md:py-24">
        <div className="grid items-center gap-10 md:grid-cols-2">
          <div>
            {eyebrow && <span className="vs-eyebrow">{eyebrow}</span>}
            <h1 className="mt-3 text-4xl md:text-5xl font-extrabold leading-tight">
              {title} {accent && <span className="text-[var(--vs-primary-600)]">{accent}</span>}
            </h1>
            {subtitle && (
              <p className="mt-4 text-[var(--vs-muted)] max-w-xl">{subtitle}</p>
            )}
            {ctas.length > 0 && (
              <div className="mt-6 flex flex-wrap gap-3">
                {ctas.map((c) => (
                  <a
                    key={c.label}
                    href={c.to}
                    className={
                      c.variant === "outline"
                        ? "vs-btn vs-btn-outline"
                        : "vs-btn vs-btn-primary"
                    }
                  >
                    {c.label}
                  </a>
                ))}
              </div>
            )}
            {/* quick stats line */}
            <div className="mt-8 grid grid-cols-3 gap-4 max-w-md text-center">
              {[
                ["Cost savings", "60%"],
                ["First reply", "<60s"],
                ["Coverage", "24/7"]
              ].map(([k, v], i) => (
                <div key={i} className="vs-card p-3">
                  <div className="text-[11px] text-slate-500">{k}</div>
                  <div className="text-lg font-bold">{v}</div>
                </div>
              ))}
            </div>
          </div>

          {rightPanel && (
            <div className="relative">
              {image ? (
                <img
                  src={image}
                  alt={imageAlt || ""}
                  className="hero-img reveal reveal-fade is-visible"
                />
              ) : (
                <div className="hero-panel reveal reveal-right is-visible">
                  <div className="hero-panel-grid" />
                  <div className="hero-badge">Agent console</div>
                  <div className="hero-chip" style={{ top: 26, left: 26 }} />
                  <div className="hero-chip" style={{ top: 80, right: 40 }} />
                  <div className="hero-chip" style={{ bottom: 36, left: 60 }} />
                </div>
              )}
            </div>
          )}
        </div>
      </div>
    </section>
  );
}
```
